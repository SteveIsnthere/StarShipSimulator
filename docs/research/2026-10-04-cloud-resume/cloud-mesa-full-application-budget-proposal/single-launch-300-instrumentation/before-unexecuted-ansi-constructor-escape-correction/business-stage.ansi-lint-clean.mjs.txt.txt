function inventory(root, knownBrowser = false) {
  const files = [], directories = []; let totalBytes = 0;
  function walk(path) {
    const s = lstatSync(path); assert(s.isDirectory() && !s.isSymbolicLink(), 'Directory identity');
    directories.push([relative(root, path), s.mode & 0o7777]);
    for (const name of readdirSync(path).sort()) {
      const p = resolve(path, name), row = lstatSync(p); assert(!row.isSymbolicLink(), 'Output/tree symlink');
      if (row.isDirectory()) walk(p);
      else {
        assert(row.isFile(), 'Output/tree special entry');
        assert(files.length < 10000 && (row.size <= 64 * 1024 * 1024 ||
          knownBrowser && p === m.browser && row.size === 290614600 &&
          digest(p) === '0b20b130e7edd9dd51873be867761295fe0cfad490c2b9a64f95bd3cfc08fa71'), 'Finite file inventory');
        totalBytes += row.size; assert(totalBytes <= 512 * 1024 * 1024, 'Finite complete tree');
        files.push([relative(root, p), digest(p), row.mode & 0o7777, row.size]);
      }
    }
  }
  walk(root); files.sort((a, b) => a[0].localeCompare(b[0])); directories.sort((a, b) => a[0].localeCompare(b[0]));
  return { files, directories, totalBytes };
}

def stat_row(pid):
    try:
        text = pathlib.Path('/proc', str(pid), 'stat').read_text()
        f = text[text.rfind(')') + 2:].split()
        return {'pid': int(pid), 'state': f[0], 'parent': int(f[1]), 'group': int(f[2]), 'session': int(f[3]), 'start': f[19]}
    except (FileNotFoundError, ProcessLookupError):
        return None

def admit(row):
    global audit_bytes
    key = (row['pid'], row['start'])
    if key not in known:
        group = owned_groups.get((row['group'], row['session']))
        require(group is None or not group['uncertain'], 'No new admission into uncertain group')
        known[key] = row
        if row['pid'] == row['group'] == row['session']:
            group_key = (row['group'], row['session'])
            if group_key in owned_groups:
                require(owned_groups[group_key]['start'] == row['start'], 'Owned group identity recycled')
            else:
                owned_groups[group_key] = {'group': row['group'], 'session': row['session'], 'start': row['start'], 'cleared': False, 'uncertain': False}
        line = json.dumps({'observedAt': time.time(), 'identity': row}) + '\n'
        audit_bytes += len(line.encode())
        require(audit_bytes <= 1024 * 1024 and len(known) <= 4096, 'Root identity audit cap')
        with (root_receipt / 'identities.jsonl').open('a') as output:
            output.write(line); output.flush(); os.fsync(output.fileno())

def scan():
    rows = []
    for entry in pathlib.Path('/proc').iterdir():
        if entry.name.isdigit():
            try:
                row = stat_row(entry.name)
            except PermissionError:
                if any(key[0] == int(entry.name) for key in known):
                    raise RuntimeError('Owned stat inaccessible')
                continue
            if row: rows.append(row)
    if child is not None:
        leader = next((r for r in rows if r['pid'] == child.pid), None)
        captured = next((r for r in known.values() if r['pid'] == child.pid), None)
        if leader and captured: require(leader['start'] == captured['start'], 'Outer PID recycled')
    changed = True
    while changed:
        changed = False
        # A captured detached leader owns its group/session until witnessed empty.
        # Freshly re-read a current member before admitting an orphaned sibling.
        for group in list(owned_groups.values()):
            if group['cleared']: continue
            leader = next((r for r in rows if r['pid'] == group['group']), None)
            if leader: require(leader['start'] == group['start'], 'Detached group leader PID recycled')
            members = [r for r in rows if r['group'] == group['group'] and r['session'] == group['session']]
            if not members:
                group['cleared'] = True
                continue
            witness = None
            for member in members:
                captured = known.get((member['pid'], member['start']))
                if captured is None or captured['group'] != group['group'] or captured['session'] != group['session']:
                    continue
                current = stat_row(member['pid'])
                if current is None: continue
                require(current['start'] == captured['start'], 'Owned group witness PID changed')
                witness = current; break
            if witness is None:
                group['uncertain'] = True
                raise RuntimeError('Group members exist without captured live identity witness; no new admission/signals')
            require(not group['uncertain'], 'Prior group ownership uncertainty is sticky')
            require(witness['group'] == group['group'] and witness['session'] == group['session'], 'Owned group witness moved')
            require(int(witness['start']) >= int(group['start']), 'Group member predates captured group')
            for row in members:
                require(int(row['start']) >= int(group['start']), 'Group member predates captured group')
                if (row['pid'], row['start']) not in known:
                    admit(row); changed = True
        for row in rows:
            if (row['pid'], row['start']) in known: continue
            parent = next((r for r in rows if r['pid'] == row['parent']), None)
            if parent and (parent['pid'], parent['start']) in known:
                admit(row); changed = True
    return [r for r in rows if (r['pid'], r['start']) in known and r['state'] not in ('Z', 'X')] + [
        {'groupUncertain': group} for group in owned_groups.values() if group['uncertain']]

def safe_scan():
    try:
        return scan()
    except BaseException as error:
        failures.append('Owned scan failure: ' + str(error))
        live = [{'groupUncertain': group} for group in owned_groups.values() if group['uncertain']]
        for row in list(known.values()):
            try:
                actual = stat_row(row['pid'])
                if actual and actual['start'] == row['start'] and actual['state'] not in ('Z', 'X'):
                    live.append(actual)
            except BaseException as error:
                failures.append('Owned identity inaccessible: ' + str(error))
                live.append({'identityUncertain': row})
        return live

def signal_rows(rows, sig):
    for row in rows:
        if 'identityUncertain' in row or 'groupUncertain' in row: continue
        group = owned_groups.get((row['group'], row['session']))
        if group is not None and group['uncertain']: continue
        try:
            actual = stat_row(row['pid'])
            if actual is None: continue
            require(actual['start'] == row['start'], 'Owned PID identity changed before signal')
            require(actual['group'] == row['group'] and actual['session'] == row['session'], 'Owned topology changed before signal')
            if actual['state'] not in ('Z', 'X'):
                try: os.kill(actual['pid'], sig)
                except ProcessLookupError: pass
        except BaseException as error:
            failures.append('Identity-checked signal failure: ' + str(error))

def load_inner():
    global inner_receipt
    output = root_receipt / 'observer.stdout.txt'
    if not output.exists(): return
    require(output.stat().st_size <= 2*1024*1024, 'Inner output cap')
    for line in output.read_text().splitlines():
        try: value = json.loads(line)
        except json.JSONDecodeError: continue
        if not isinstance(value, dict) or not isinstance(value.get('rootReceipt'), str): continue
        candidate = pathlib.Path(value['rootReceipt'])
        require(str(candidate).startswith('/tmp/starship-original21-root.') and candidate.resolve() == candidate and candidate.is_dir(), 'Actual inner root receipt path')
        require(inner_receipt is None or inner_receipt == candidate, 'Multiple inner receipts')
        inner_receipt = candidate
    if inner_receipt is None: return
    journal = inner_receipt / 'identities.jsonl'
    if journal.exists():
        require(journal.stat().st_size <= 1024*1024, 'Inner identity journal cap')
        lines = journal.read_text().splitlines()
        for index, line in enumerate(lines):
            try: record = json.loads(line)
            except json.JSONDecodeError:
                require(index == len(lines)-1, 'Corrupt completed identity journal line'); continue
            row = record['identity']
            for key in ('pid','start','group','session'): require(key in row, 'Malformed actual inner identity')
            current = stat_row(row['pid'])
            if current:
                require(current['start'] == row['start'] and current['group'] == row['group'] and current['session'] == row['session'], 'Inner tracked identity changed; no replacement admission')
            admit(row)

def run_child(label, args, deadline):
    global child
    with (root_receipt / (label + '.stdout.txt')).open('wb') as stdout, (root_receipt / (label + '.stderr.txt')).open('wb') as stderr:
        child = subprocess.Popen(args, cwd=m['root'], env=dict(os.environ), stdout=stdout, stderr=stderr, start_new_session=True)
        initial = stat_row(child.pid)
        require(initial is not None and initial['parent'] == os.getpid() and initial['group'] == child.pid and initial['session'] == child.pid, 'Immediate direct Popen registration')
        admit(initial); save(label + '-registration.json', initial)
        while child.poll() is None:
            scan(); load_inner()
            require(time.time() <= deadline, 'Wrapper absolute deadline: ' + label)
            require(sum((root_receipt / (label + suffix)).stat().st_size for suffix in ('.stdout.txt','.stderr.txt')) <= 2*1024*1024, 'Wrapper combined live output cap')
            time.sleep(0.025)
        save(label + '-actual-exit.json', {'returncode': child.returncode, 'elapsedSeconds': time.time() - started})
        require(time.time() <= deadline, 'Wrapper post-exit deadline: ' + label)
        load_inner(); scan()
        return child.returncode

def structural(phase, deadline):
    code = run_child('structural-' + phase, [w['python']['path'], str(checker), phase], deadline)
    phase_statuses.append({'phase': phase, 'returncode': code})
    require(code == 0, 'Structural checker failed: ' + phase)
    value = json.loads((root_receipt / ('root-structural-' + phase + '.json')).read_text())
    require(value['code'] == 0 and value['phase'] == phase and value['checkerSha256'] == w['checker']['sha'] and value['closureSha256'] == w['closure']['sha'], 'Actual structural receipt mismatch')

def final_structural(deadline):
    """Reserved direct checker is independent of uncertain historical groups."""
    global final_checker, final_checker_identity
    label = 'structural-after'
    # This is a fresh detached direct Popen, never a group-discovered admission.
    # No historical scan or inner journal error may preempt its bounded polling.
    with (root_receipt / (label + '.stdout.txt')).open('wb') as stdout, (root_receipt / (label + '.stderr.txt')).open('wb') as stderr:
        require(time.time() < deadline, 'No time remaining for finally structural checker')
        final_checker = subprocess.Popen([w['python']['path'], str(checker), 'after'], cwd=m['root'], env=dict(os.environ), stdout=stdout, stderr=stderr, start_new_session=True)
        try:
            initial = stat_row(final_checker.pid)
            require(initial is not None and initial['parent'] == os.getpid() and initial['group'] == final_checker.pid and initial['session'] == final_checker.pid, 'Finally checker immediate direct Popen registration')
            final_checker_identity = initial
            admit(initial); save(label + '-registration.json', initial)
        except BaseException as error:
            failures.append('Finally checker immediate registration: ' + str(error))
        try:
            while final_checker.poll() is None:
                require(time.time() <= deadline, 'Finally checker absolute deadline')
                require(sum((root_receipt / (label + suffix)).stat().st_size for suffix in ('.stdout.txt','.stderr.txt')) <= 2*1024*1024, 'Finally checker combined output cap')
                time.sleep(0.025)
        finally:
            # Actual status is retained before any output/receipt assertion.
            try: save(label + '-actual-exit.json', {'returncode': final_checker.poll(), 'elapsedSeconds': time.time() - started})
            except BaseException as error: failures.append('Finally checker raw publication: ' + str(error))
        require(time.time() <= deadline, 'Finally checker post-exit deadline')
        require(sum((root_receipt / (label + suffix)).stat().st_size for suffix in ('.stdout.txt','.stderr.txt')) <= 2*1024*1024, 'Finally checker final output cap')
        code = final_checker.returncode
        phase_statuses.append({'phase': 'after', 'returncode': code})
        require(code == 0, 'Finally structural checker failed')
        value = json.loads((root_receipt / 'root-structural-after.json').read_text())
        require(value['code'] == 0 and value['phase'] == 'after' and value['checkerSha256'] == w['checker']['sha'] and value['closureSha256'] == w['closure']['sha'], 'Actual finally structural receipt mismatch')

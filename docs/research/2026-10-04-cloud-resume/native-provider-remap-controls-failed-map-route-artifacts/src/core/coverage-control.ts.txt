export enum ControlReason { No = 0, Yes = 1 }
export function choose(value: number): number {
  if (value > 0) return ControlReason.Yes;
  return ControlReason.No;
}
export function nested(value: number): number {
  if (value > 0) {
    if (value > 10) return 10;
    return 1;
  }
  return 0;
}
export function neverCalled(value: number): number {
  return value > 0 ? 3 : 4;
}

pins_path=Path(sys.argv[1]);assert sha(pins_path.read_bytes())==sys.argv[2]

def durable(p,b):
    with p.open('xb') as f:f.write(b);f.flush();os.fsync(f.fileno())
    p.chmod(0o600)
    fd=os.open(p.parent,os.O_RDONLY|os.O_DIRECTORY)
    try:os.fsync(fd)
    finally:os.close(fd)

def read(p):
    assert p.is_absolute()
    for a in [p,*p.parents]:assert not a.is_symlink()
    st=p.lstat();assert stat.S_ISREG(st.st_mode)
    cap=16777216
    if str(p.relative_to(root))==plan['exactPlainPatchException']['path']:
        cap=18837675;assert st.st_size==cap
    assert st.st_size<=cap
    b=p.read_bytes();assert len(b)==st.st_size
    if cap==18837675:assert sha(b)==plan['exactPlainPatchException']['sha256']
    return b,stat.S_IMODE(st.st_mode)

copyrows=[]
for row in plan['losslessCopies']:
    original=root/row['original'];target=root/row['canonical'];b,mode=read(original)
    assert target.parent==original.parent and target.name==original.name+'.txt'
    created=False
    if os.path.lexists(target):assert read(target)[0]==b
    else:durable(target,b);created=True
    assert read(original)==(b,mode) and read(target)[0]==b
    copyrows.append({**row,'bytes':len(b),'sha256':sha(b),'originalMode':mode,'canonicalMode':stat.S_IMODE(target.lstat().st_mode),'created':created})
durable(setup/'lossless-copy-origins.json',(json.dumps(copyrows,indent=2)+'\n').encode())

for row in copyrows:assert read(root/row['original'])==(read(root/row['canonical'])[0],row['originalMode'])

def bound():

def save(name,b):
    p=output/name
    with p.open('xb') as f:f.write(b);f.flush();os.fsync(f.fileno())
    p.chmod(0o600)
    fd=os.open(output,os.O_RDONLY|os.O_DIRECTORY)
    try:os.fsync(fd)
    finally:os.close(fd)

def proc(pid):
    try:
        s=Path('/proc',str(pid),'stat').read_text();f=s[s.rfind(')')+2:].split()
        return {'pid':pid,'state':f[0],'parent':int(f[1]),'group':int(f[2]),'session':int(f[3]),'start':f[19]}
    except (FileNotFoundError,ProcessLookupError):return None

def terminate(c,identity):
    if c.poll() is not None:return
    now=proc(c.pid)
    assert identity is not None and now is not None and now['start']==identity['start']
    assert now['parent']==os.getpid() and now['group']==now['session']==c.pid
    os.kill(c.pid,signal.SIGTERM)
    try:c.wait(timeout=.2)
    except subprocess.TimeoutExpired:
        now=proc(c.pid);assert now is not None and now['start']==identity['start']
        os.kill(c.pid,signal.SIGKILL);c.wait(timeout=1)

def git(args):
    bound(); n=len(commands); prefix=str(n)
    allowed=[['rev-parse','HEAD'],['symbolic-ref','--short','HEAD'],['diff','--cached','--name-only','-z'],['status','--porcelain=v1','-z','--untracked-files=all']]
    assert args in allowed
    save(prefix+'.started.json',(json.dumps({'args':args})+'\n').encode())
    env=dict(os.environ);env['GIT_OPTIONAL_LOCKS']='0'
    c=None;identity=None;error=None;streams={};sel=selectors.DefaultSelector();total=0;result=None;truncated=False
    try:
        c=subprocess.Popen([m['git']['path'],*args],cwd=root,env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE,start_new_session=True)
        for label,stream in [('stdout',c.stdout),('stderr',c.stderr)]:
            os.set_blocking(stream.fileno(),False)
            path=output/(prefix+'.'+label+'.bin');f=path.open('xb');path.chmod(0o600)
            streams[label]=f;sel.register(stream,selectors.EVENT_READ,label)
        identity=proc(c.pid)
        assert identity is not None and identity['parent']==os.getpid() and identity['group']==identity['session']==c.pid
        save(prefix+'.identity.json',(json.dumps(identity)+'\n').encode())
        deadline=time.monotonic()+5
        while sel.get_map():
            bound();assert time.monotonic()<deadline, 'Git command deadline'
            for key,_ in sel.select(.02):
                data=os.read(key.fileobj.fileno(),8192)
                if not data:sel.unregister(key.fileobj);continue
                remaining=2097152-total
                if len(data)>remaining:
                    if remaining:streams[key.data].write(data[:remaining]);total+=remaining
                    truncated=True;raise RuntimeError('Combined streamed Git output cap')
                streams[key.data].write(data);total+=len(data)
        c.wait(timeout=max(.001,deadline-time.monotonic()))
    except BaseException:error=traceback.format_exc()
    finally:
        if c is not None:
            # Independent identity recovery only for an unreaped direct child reserved by Popen.
            try:
                if identity is None and c.poll() is None:
                    now=proc(c.pid)
                    assert now is not None and now['parent']==os.getpid() and now['group']==now['session']==c.pid
                    identity=now;save(prefix+'.fallback-identity.json',(json.dumps(identity)+'\n').encode())
                terminate(c,identity)
                if c.poll() is None:c.wait(timeout=1)
            except BaseException:error=(error or '')+traceback.format_exc()
        for f in streams.values():
            try:f.flush();os.fsync(f.fileno());f.close()
            except BaseException:error=(error or '')+traceback.format_exc()
        for stream in ([] if c is None else [c.stdout,c.stderr]):
            if stream is not None:stream.close()
        sel.close()
        result={'args':args,'pidStart':identity,'exitCode':None if c is None else c.returncode,'failure':error,'truncated':truncated,'retainedBytes':total}
        commands.append(result);save(prefix+'.completed.json',(json.dumps(result)+'\n').encode())
    assert error is None and c.returncode==0 and not truncated
    return (output/(prefix+'.stdout.bin')).read_bytes()

def boundary():
    head=git(['rev-parse','HEAD']).decode().strip()
    branch=git(['symbolic-ref','--short','HEAD']).decode().strip()
    staged=git(['diff','--cached','--name-only','-z'])
    status=git(['status','--porcelain=v1','-z','--untracked-files=all'])
    p=root/'.git/index';st=p.lstat();assert stat.S_ISREG(st.st_mode) and st.st_size<=8388608
    return {'head':head,'branch':branch,'staged':staged.hex(),'statusSha256':sha(status),'indexSha256':sha(p.read_bytes()),'indexMode':stat.S_IMODE(st.st_mode)}

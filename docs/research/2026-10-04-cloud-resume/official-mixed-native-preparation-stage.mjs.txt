import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, mkdtempSync, copyFileSync, chmodSync, realpathSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { resolve } from 'node:path';
import { pathToFileURL } from 'node:url';
const sha = file => createHash('sha256').update(readFileSync(file)).digest('hex');
const [manifestPath, manifestSha] = process.argv.slice(2);
assert.match(manifestSha, /^[a-f0-9]{64}$/); assert.equal(sha(manifestPath), manifestSha);
const m = JSON.parse(readFileSync(manifestPath, 'utf8'));
assert.equal(m.kind, 'official-mixed-native-preparation-owner-input-v1');
assert.equal(process.cwd(), realpathSync(m.root)); assert.equal(process.version, 'v22.23.3');
assert.equal(sha(process.execPath), m.node.sha);
for (const key of ['helper', 'utility', 'preparer', 'pins', 'npmCli', 'stage', 'observer', 'python']) {
  assert.equal(sha(m[key].path), m[key].sha, key);
}
const deadline = Number(process.env.OFFICIAL_NATIVE_ROOT_DEADLINE);
assert(Number.isFinite(deadline) && deadline > Date.now() && deadline <= Date.now() + 90000);
const stage = mkdtempSync('/tmp/starship-official-native-stage.'); chmodSync(stage, 0o700);
process.stdout.write(JSON.stringify({ stageReceipt: stage, manifestSha }) + '\n');
const preparer = resolve(stage, 'prepare.mjs'); copyFileSync(m.preparer.path, preparer);
assert.equal(sha(preparer), m.preparer.sha);
const { runOwnedCommand, verifyNoOwnedProcesses } = await import(pathToFileURL(m.helper.path).href);
let code = null, failure = null, actualReceipt = null;
try {
  code = await runOwnedCommand({ name: 'preparation',
    args: [preparer, m.root, m.pins.path, m.pins.sha, m.utility.sha, m.helper.sha, m.node.sha, m.npmCli.path],
    root: m.root, receipt: stage, env: { ...process.env }, deadlineMs: deadline - Date.now() });
} catch (error) { failure = error; }
// Preserve actual status BEFORE asserting any produced receipt.
let actualExit = null;
try { actualExit = JSON.parse(readFileSync(resolve(stage, 'preparation.process.json'), 'utf8')).exit; }
catch (error) { failure ??= error; }
writeFileSync(resolve(stage, 'actual-child-exit.json'), JSON.stringify({ code, exit: actualExit,
  error: failure ? String(failure.stack ?? failure) : null,
  processReceipt: resolve(stage, 'preparation.process.json') }) + '\n', { flag: 'wx' });
try {
  const processReceipt = JSON.parse(readFileSync(resolve(stage, 'preparation.process.json'), 'utf8'));
  assert.equal(code, 0); assert.equal(processReceipt.exit.code, 0);
  assert.equal(processReceipt.exit.signal, null); assert.equal(processReceipt.failure, null);
  assert.equal(processReceipt.finished, true); assert.equal(processReceipt.closed, true);
  assert.equal(processReceipt.cleanup.executing.length, 0);
  const rows = readFileSync(resolve(stage, 'preparation.stdout.txt'), 'utf8').trim().split(/\r?\n/).map(line => JSON.parse(line));
  assert(rows.length > 0);
  const locations = [...new Set(rows.map(row => row.receipt).filter(value => typeof value === 'string'))];
  assert.equal(locations.length, 1); actualReceipt = locations[0];
  assert.match(actualReceipt, /^\/tmp\/starship-official-native\.[A-Za-z0-9]+$/);
  assert.equal(realpathSync(actualReceipt), actualReceipt);
  const resultPath = resolve(actualReceipt, 'preparation-result.json');
  const result = JSON.parse(readFileSync(resultPath, 'utf8'));
  assert.equal(result.completed, true); assert.equal(result.candidateQualified, false);
  assert.equal(result.receipt, actualReceipt); assert.equal(sha(result.inputPath), result.inputSha);
  assert.equal(result.nodeBinary, m.node.sha); assert.equal(result.pinsSha, m.pins.sha);
  assert.equal(sha(manifestPath), manifestSha);
  for (const key of ['helper', 'utility', 'preparer', 'pins', 'npmCli', 'stage', 'observer', 'python']) assert.equal(sha(m[key].path), m[key].sha);
  assert.equal(sha(process.execPath), m.node.sha);
  assert(verifyNoOwnedProcesses().every(row => row.executing.length === 0));
  assert(Date.now() <= deadline, 'Preparation workload exceeded root90s bound');
} catch (error) { failure ??= error; }
writeFileSync(resolve(stage, 'stage-result.json'), JSON.stringify({ completed: !failure,
  code, error: failure ? String(failure.stack ?? failure) : null, manifestSha, actualReceipt,
  processReceipt: resolve(stage, 'preparation.process.json'),
  acceptance: 'Input evidence only; root observer actual cleanup and independent input review required' }) + '\n', { flag: 'wx' });
process.stdout.write(JSON.stringify({ stageReceipt: stage, actualReceipt, completed: !failure }) + '\n');
if (failure) throw failure;

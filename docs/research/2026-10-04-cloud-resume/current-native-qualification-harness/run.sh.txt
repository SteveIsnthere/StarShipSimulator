#!/usr/bin/env bash
set -euo pipefail
routes=(/workspace/cloud-bootstrap/starship-v*/activate.sh)
[[ ${#routes[@]} == 1 && -f ${routes[0]} ]] || exit 69
source "${routes[0]}"
cd /workspace/StarShipSimulator
[[ ${1:-} =~ ^(kernels|paired|cost)$ ]] || exit 69
[[ ${STARSHIP_CPU_SLOT_GRANTED:-} == current-native-$1 ]] || exit 69
[[ -n ${STARSHIP_CURRENT_NATIVE_REVIEW_SHA:-} && -n ${STARSHIP_CURRENT_NATIVE_REVIEW_PINS_SHA:-} ]] || exit 69
export npm_execpath
npm_execpath=$(readlink -f "$(command -v npm)")
draft=docs/research/2026-10-04-cloud-resume/current-native-qualification-harness
# Own launcher bytes are retained, checked against the exact template transformation before/after.
launchdir=$(mktemp -d "$draft/owned-launch-XXXXXXXX")
python3 - "$draft" "$launchdir" <<'PY'
from pathlib import Path
import sys,json,hashlib
r=Path(sys.argv[1]);out=Path(sys.argv[2])
source=(r/'launcher.mjs.txt').read_text()
for token,path in [('NATIVE_LOADER_PLACEHOLDER',Path('scripts/bench/native-loader.mjs').resolve()),('OWNED_COMMAND_PLACEHOLDER',(out/'owned-command.mjs').resolve())]:
 assert source.count(token)==1
 source=source.replace(token,str(path))
(out/'launcher.mjs').write_text(source)
(out/'owned-command.mjs').write_bytes((r/'owned-command.mjs.txt').read_bytes())
(out/'executed-before.json').write_text(json.dumps({str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in [out/'launcher.mjs',out/'owned-command.mjs']},indent=2)+'\n')
PY
status=0
node "$launchdir/launcher.mjs" "$@" > "$launchdir/stdout.txt" 2> "$launchdir/stderr.txt" || status=$?
printf '%s\n' "$status" > "$launchdir/node-status.txt"
python3 - "$launchdir" <<'PY' || status=70
from pathlib import Path
import json,hashlib,sys
r=Path(sys.argv[1]);before=json.loads((r/'executed-before.json').read_text())
after={p:hashlib.sha256(Path(p).read_bytes()).hexdigest() for p in before}
(r/'executed-after.json').write_text(json.dumps(after,indent=2)+'\n')
assert after==before,'actual executed launcher/owner bytes changed'
PY
printf '%s\n' "$launchdir"
exit "$status"

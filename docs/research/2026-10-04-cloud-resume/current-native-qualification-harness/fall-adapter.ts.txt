function cloneData(value:unknown,seen=new WeakMap<object,object>()):any{
 if(typeof value==='function')throw new Error('unsupported opaque function in fall recipe');
 if(!value||typeof value!=='object')return value;
 if(seen.has(value))return seen.get(value);
 ok(Object.getPrototypeOf(value)===Object.prototype||Object.getPrototypeOf(value)===Array.prototype||Object.getPrototypeOf(value)===null,'unsupported fall recipe prototype');
 const out=Array.isArray(value)?[]:{};seen.set(value,out);
 for(const key of Reflect.ownKeys(value)){
  ok(typeof key==='string','fall recipe symbol');const descriptor=Object.getOwnPropertyDescriptor(value,key)!;ok('value'in descriptor,'fall recipe accessor');
  if(Array.isArray(value)&&key==='length')continue;
  Object.defineProperty(out,key,{...descriptor,value:cloneData(descriptor.value,seen)});
 }
 if(Object.isFrozen(value))Object.freeze(out);return out;
}

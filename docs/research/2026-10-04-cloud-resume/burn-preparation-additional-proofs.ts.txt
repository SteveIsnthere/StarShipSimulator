// Example full-mock collection (separate .test.ts after freeze) must dynamically

import { expect, it, vi } from 'vitest';

it('a full propulsion mock with no certificate retains the original helper path', async () => {
  vi.resetModules();
  const original = await vi.importActual<typeof import('$core/physics/propulsion')>('$core/physics/propulsion');
  const flow = vi.fn(original.engineMassFlow), thrust = vi.fn(original.engineThrust);
  vi.doMock('$core/physics/propulsion', () => ({ engineMassFlow: flow, engineThrust: thrust }));
  try {
    const current = await import('$core/control/guidance-physics');
    const before = await import('../../../tests/proofs/fixtures/burn-before-preparation/guidance-physics');
    const { SUPER_HEAVY } = await import('$core/vehicles/super-heavy');
    const namespace = await import('$core/physics/propulsion');
    expect('isCanonicalBurnPropulsion' in namespace).toBe(false);
    const actual = current.createBurnScratch(), expected = before.createBurnScratch();
    const got = current.landingBurnStartAltitude(3, 330000, 30, 90.5, actual, SUPER_HEAVY);
    const want = before.landingBurnStartAltitude(3, 330000, 30, 90.5, expected, SUPER_HEAVY);
    expect(Object.is(got, want)).toBe(true);
    // In the real proof use recursive exact(Object.is) for EVERY scratch leaf,
    // retain identities, and mirrored old helper sequence rather than toEqual.
    expect(thrust.mock.calls.length).toBeGreaterThan(2);
    expect(flow.mock.calls.length).toBeGreaterThan(1);
  } finally {
    vi.doUnmock('$core/physics/propulsion'); vi.resetModules();
  }
});

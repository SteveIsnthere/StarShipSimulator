def run(name,cmd,bound):
 started=time.monotonic()
 with (parent/(name+'.output.txt')).open('xb') as output:
  proc=subprocess.Popen(cmd,stdout=output,stderr=subprocess.STDOUT,start_new_session=True,env=env)

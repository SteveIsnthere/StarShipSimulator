def durable(p,b):
    bound()
    with p.open('xb') as f:f.write(b);f.flush();os.fsync(f.fileno())
    p.chmod(0o600)
    fd=os.open(p.parent,os.O_RDONLY|os.O_DIRECTORY)
    try:os.fsync(fd)
    finally:os.close(fd)

def private_environment(value):
    # Inspect structured data only; do not print values or treat source text as instructions.
    stack=[value]
    while stack:
        v=stack.pop()
        if isinstance(v,dict):
            browser=v.get('browser')
            if isinstance(browser,dict) and isinstance(browser.get('launch'),dict):
                assert browser['launch'].get('env') in (None,{}), 'Private browser launch environment cannot be published'
            stack.extend(v.values())
        elif isinstance(v,list):stack.extend(v)

def privacy(b):
    stripped=b.lstrip()
    if stripped[:1] not in (b'{',b'['):return
    try:value=json.loads(b)
    except (json.JSONDecodeError,UnicodeDecodeError):
        assert not (b'"browser"' in b and b'"env"' in b), 'Unclassifiable structured environment input needs safe omission'
        return
    private_environment(value)

/** Outer validator only AFTER declared child exit and before/after pin checks. */
import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {fileURLToPath} from 'node:url';
const dir=process.argv[2],out=process.argv[3],bridgePath=process.argv[4];if(!dir||!out||!bridgePath)throw Error('Exact immutable episode, private offline output and reviewed bridge required');
const bridge=JSON.parse(fs.readFileSync(bridgePath,'utf8'));
const digest=p=>createHash('sha256').update(fs.readFileSync(p)).digest('hex');
assert.equal(dir,bridge.episode);assert.equal(digest(fileURLToPath(import.meta.url)),bridge.replacement.sha256);
for(const [path,hash]of Object.entries(bridge.inputs))assert.equal(digest(path),hash,path);
const archive=JSON.parse(fs.readFileSync(`${dir}/executed-source-archive-map.json`,'utf8'));
const archived=new Map(archive.files.map(e=>[e.originalPath,e]));
const originalPins=JSON.parse(fs.readFileSync(`${dir}/before-pins.json`,'utf8'));
const absent=path=>{try{fs.lstatSync(path);assert.fail(`Recorded absent path exists:${path}`);}catch(error){if(error.code!=='ENOENT')throw error;}};
const retired=bridge.retiredMaterialization;
assert.ok(retired);assert.equal(retired.path,'/workspace/StarShipSimulator/node_modules/.cache/starship-duplicate-flight-repair1-owner.mjs');
const declaration=JSON.parse(fs.readFileSync(retired.declaration,'utf8')),result=JSON.parse(fs.readFileSync(retired.result,'utf8'));
assert.equal(declaration.executedPath,retired.path);assert.equal(declaration.sha256,retired.sha256);assert.equal(declaration.oneInvocation,true);
assert.equal(result.executedBytesStillExact,true);assert.equal(result.actualChildReturnCode,1);
assert.equal(digest(retired.reviewedTemplate),retired.sha256);assert.equal(retired.sha256,originalPins.expected['duplicate-flight-lifecycle-outer.mjs.txt']);
assert.equal(originalPins.hashes[retired.path],retired.sha256);absent(retired.path);
for(const [path,hash]of Object.entries(originalPins.hashes)){
 if(hash===null){absent(path);continue;}
 if(path===retired.path)continue;
 const entry=archived.get(path);assert.equal(digest(entry?.archivePath??path),hash,path);if(entry)assert.equal(entry.sha256,hash,path);
}
assert.equal(digest(`${dir}/finalize.mjs.txt`),bridge.originalFinalizer.sha256);
assert.equal(bridge.originalFinalizer.sha256,originalPins.executed['finalize.mjs'].sha256);
assert.equal(fs.existsSync(out),false);fs.mkdirSync(out,{mode:0o700});
const read=n=>JSON.parse(fs.readFileSync(`${dir}/${n}`,'utf8'));
assert.deepEqual(read('before-pins.json'),read('after-pins.json'));
assert.deepEqual(read('runner-closed.json'),{closed:true,unhandledErrors:0,fileErrors:0,nativeExitCode:1});
assert.equal(fs.existsSync(`${dir}/runner-failure.json`),false);
const termination=read('termination.json');assert.deepEqual(termination.exit,{code:1,signal:null});assert.equal(termination.timedOut,false);assert.equal(termination.verifiedOwnedAbsence,true);assert.deepEqual(termination.remaining,[]);assert.deepEqual(termination.errors,[]);assert.ok(termination.cleanupMs<=5000);
const pins=read('before-pins.json');for(const [actual,value]of Object.entries(pins.executed))assert.equal(value.sha256,pins.expected[value.template],actual);
assert.equal(fs.readFileSync(`${dir}/exit-status.txt`,'utf8').trim(),'1');
const report=read('test-results.json'),raw=read('raw-observations.json');
const fail=new Set(['body-fail','afterEach-fail','finished-fail','aroundEach-fail','fixture-fail','timeout']);
const names=['pass','body-fail','afterEach-fail','finished-fail','aroundEach-fail','fixture-fail','timeout','dynamic-skip','expected-failure','retry','repeat','concurrent'];
assert.equal(report.numTotalTests,24);assert.equal(report.numFailedTests,6);assert.equal(report.numPendingTests,1);assert.equal(report.numPassedTests,17);
assert.equal(report.testResults.length,1);
const tests=report.testResults.flatMap(f=>f.assertionResults);
assert.equal(tests.length,24);
assert.deepEqual([...tests.map(t=>t.fullName)].sort(),names.flatMap(n=>[`${n} producer`,`${n} consumer`]).sort());
for(const kind of names){
 const producer=tests.find(t=>t.fullName===`${kind} producer`),consumer=tests.find(t=>t.fullName===`${kind} consumer`);
 assert.equal(consumer.status,'passed');assert.equal(producer.status,fail.has(kind)?'failed':kind==='dynamic-skip'?'skipped':'passed');
 const row=raw.find(r=>r.kind===kind);assert.ok(row);assert.equal(row.runs,kind==='pass'?1:kind==='retry'||kind==='repeat'?3:2);
 assert.equal(row.producer.result.state,fail.has(kind)?'fail':kind==='dynamic-skip'?'skip':'pass');
}
assert.equal(raw.length,12);
// Intentional exit1 alone is never proof. This wx receipt exists only after ALL
// child/report/lifecycle/counter/pin checks have completed in the outer process.
fs.writeFileSync(`${out}/offline-validated-proof.json`,JSON.stringify({publicRunnerLifecycleValidated:true,childExit:1,tests:24,consumers:12,productFlightAcceptance:false},null,2)+'\n',{flag:'wx'});

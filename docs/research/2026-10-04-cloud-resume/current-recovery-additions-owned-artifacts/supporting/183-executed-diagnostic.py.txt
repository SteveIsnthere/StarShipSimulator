def require(ok, message):
    if not ok: raise RuntimeError(message)
def sha(path):
    path = pathlib.Path(path)
    require(path.is_file() and not path.is_symlink(), 'Regular selected source/tool')
    size=path.stat().st_size
    exact_git = str(path)=='/usr/local/bin/git' and size==19295504
    exact_python = str(path)=='/opt/codex/runtimes/codex-primary-runtime/dependencies/python/bin/python3.12' and size==30894944
    require(size<=8*1024*1024 or exact_git or exact_python, 'Finite selected metadata/source; only exact root-pinned Git and managed Python binaries excepted')
    value=hashlib.sha256(path.read_bytes()).hexdigest()
    if exact_git: require(value=='37d928337f8e6c6b4c875d79fd33ee2f440594a589f584c89146bbf0aeafd63e', 'Exact root-selected Git binary')
    if exact_python: require(value=='fa67443527ed9647f760d807e2a38f26340757123e643c4639cf273ed15d5ea7', 'Exact root-selected managed Python binary')
    return value

def save(name, value):
    with (receipt/name).open('x') as output:
        json.dump(value,output,indent=2); output.write('\n'); output.flush(); os.fsync(output.fileno())
def row(pid):
    text = pathlib.Path('/proc',str(pid),'stat').read_text(); fields=text[text.rfind(')')+2:].split()
    return {'pid':pid,'start':fields[19],'parent':int(fields[1]),'group':int(fields[2]),'session':int(fields[3]),'state':fields[0]}
def run(name,args):
    global total_bytes
    require(time.monotonic() < deadline, 'Absolute10s cohort bound before launch')
    command=[m['git']['path'],*args]; save(name+'.request.json',{'command':command,'cwd':str(root)})
    child=subprocess.Popen(command,cwd=root,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    selector=selectors.DefaultSelector(); streams={}; failure=None; observed=0; code=None; saved=None
    try:
        saved=row(child.pid); require(saved['parent']==os.getpid(),'Reserved direct Popen child')
        identities.append(saved); save(name+'.identity.json',saved)
        with (receipt/'identities.jsonl').open('a') as audit:
            audit.write(json.dumps(saved)+'\n'); audit.flush(); os.fsync(audit.fileno())
        for stream_name,pipe in (('stdout',child.stdout),('stderr',child.stderr)):
            streams[stream_name]=(receipt/(name+'.'+stream_name+'.raw')).open('xb')
            os.set_blocking(pipe.fileno(),False); selector.register(pipe,selectors.EVENT_READ,stream_name)
        while selector.get_map():
            require(time.monotonic()<deadline,'Absolute10s cohort output deadline')
            for key,_ in selector.select(min(.025,max(0,deadline-time.monotonic()))):
                block=os.read(key.fileobj.fileno(),min(65536,cap-total_bytes+1))
                if not block: selector.unregister(key.fileobj); continue
                room=max(0,cap-total_bytes); streams[key.data].write(block[:room]); streams[key.data].flush(); os.fsync(streams[key.data].fileno())
                observed+=len(block); total_bytes+=len(block); require(total_bytes<=cap,'Combined cohort8MiB cap exceeded; raw prefix only')
        code=child.wait(timeout=max(.001,deadline-time.monotonic()))
    except BaseException as error:
        failure=error
    finally:
        # Independently attempt every teardown stage; uncertainty is always red.
        teardown_errors=[]
        # The unreaped direct Popen PID stays reserved until this exact child is reaped.
        try:
            if child.poll() is None:
                live=row(child.pid); require(live['parent']==os.getpid(),'Direct Popen child parent mismatch before kill')
                if saved: require(all(live[k]==saved[k] for k in ('start','group','session')),'Direct Popen identity changed before kill')
                child.kill()
        except BaseException as error: teardown_errors.append({'stage':'identity-checked-kill','error':str(error)})
        try: code=child.wait(timeout=1)
        except BaseException as error: teardown_errors.append({'stage':'exact-Popen-wait','error':str(error)})
        # Persist actual OS status before subsequent policy or pipe/file assertions.
        try:
            save(name+'.exit.json',{'returncode':child.returncode,'identity':saved,'observedBytes':observed,'combinedCohortBytes':total_bytes,
                 'truncated':total_bytes>cap,'primaryError':None if failure is None else str(failure),
                 'teardownErrorsAtExit':list(teardown_errors),'elapsedSeconds':time.monotonic()-started})
        except BaseException as error: teardown_errors.append({'stage':'actual-exit-fsync','error':str(error)})
        for label,closer in [('selector-close',selector.close),('stdout-pipe-close',child.stdout.close),('stderr-pipe-close',child.stderr.close)]:
            try: closer()
            except BaseException as error: teardown_errors.append({'stage':label,'error':str(error)})
        for label,output in streams.items():
            try: output.close()
            except BaseException as error: teardown_errors.append({'stage':label+'-raw-file-close','error':str(error)})
        try:
            save(name+'.teardown.json',{'returncode':child.returncode,'primaryError':None if failure is None else str(failure),
                 'teardownErrors':teardown_errors,'identity':saved})
        except BaseException as error: teardown_errors.append({'stage':'teardown-fsync','error':str(error)})
    if failure is not None or teardown_errors:
        raise RuntimeError(json.dumps({'primaryError':None if failure is None else str(failure),'teardownErrors':teardown_errors})) from failure
    stdout=receipt/(name+'.stdout.raw'); stderr=receipt/(name+'.stderr.raw')
    commands.append({'name':name,'code':code,'stdoutBytes':stdout.stat().st_size,'stderrBytes':stderr.stat().st_size,
                     'stdoutSha':sha(stdout),'stderrSha':sha(stderr),'identity':saved})
    require(code==0 and stderr.stat().st_size==0,'Actual Git failure/stderr: '+name)
    require(time.monotonic()<deadline,'Absolute10s cohort post-exit bound')
    return stdout.read_bytes()

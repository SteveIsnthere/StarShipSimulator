function errorSnapshot(error){
 const keys=Reflect.ownKeys(error);assert(keys.every(key=>typeof key==='string'));
 return {name:error.name,message:error.message,code:error.code,operator:error.operator,
  actual:error.actual,expected:error.expected,ownKeys:keys,
  descriptors:Object.fromEntries(keys.map(key=>{const d=Object.getOwnPropertyDescriptor(error,key);
   assert('value'in d,'Error accessor unsupported');return[key,{...d,value:key==='stack'?'SOURCE_STACK_EXCLUDED':d.value}];}))};
}
function parityCase(name,recipe,expectedMessage){
 test(name,()=>{
  const execute=make=>{const log=[],fixture=recipe(log);let o,error;
   try{o=make(...(fixture.api??apis()));fixture.run(o);}catch(caught){error=caught;}
   assert(error instanceof Error,'Negative must throw');
   if(expectedMessage!==undefined)assert.equal(error.message,expectedMessage);
   return{snapshot:errorSnapshot(error),constructor:error.constructor,counts:o?.counts()??null,log};};
  const before=execute(beforeOracle),after=execute(graphOracle);
  assert.equal(after.constructor,before.constructor);assert.deepEqual(after.snapshot,before.snapshot);
  assert.deepEqual(after.counts,before.counts);assert.deepEqual(after.log,before.log);
  errorEvidence.push({name,before:before.snapshot,after:after.snapshot,beforeCounts:before.counts,afterCounts:after.counts,log:before.log});
 });
}

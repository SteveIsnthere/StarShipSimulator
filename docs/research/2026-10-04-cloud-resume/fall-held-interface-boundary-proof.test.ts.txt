/** Research candidate proof: retained independent continuation is never edited. */
import {describe,it,expect,vi} from 'vitest';
import {advanceUnpoweredFall,createBurnScratch,createFallResult,createUnpoweredFallWork,unpoweredFallInto,type UnpoweredFallWork} from '$core/control/guidance-physics';
import {advancePrototypeFall,createPrototypeScratch,prototypeFallInto} from './PROTOTYPE_PLACEHOLDER';
import {createScenarioState,getScenario} from '$core/scenarios';

import {SHIP} from '$core/vehicle';

function source(booster=false) {
 const model=booster?SUPER_HEAVY:SHIP;
 const state=booster?createInitialState(123,model):createScenarioState(getScenario('reentry')!);
 Object.assign(state.kinematics,{altitude:79000,speedX:2000,speedY:-50});
 state.vehicle.frontFinExtension=booster?50:0;state.vehicle.aftFinExtension=0;
 return {state,model};
}
function checkpoint(work:UnpoweredFallWork) {
 return {state:work.state,model:work.model,groundAltitude:work.groundAltitude,pitch:work.pitch,
  mass:work.mass,maxArea:work.maxArea,referenceWind:work.referenceWind,
  zeroControlArea:work.zeroControlArea,zeroControlColumnArea:work.zeroControlColumnArea,
  controlModel:work.controlModel,h:work.h,x:work.x,vx:work.vx,vy:work.vy,
  steps:work.steps,done:work.done,result:work.result};
}
function caught(run:()=>unknown) {
 try {return {used:run(),error:null};}
 catch(error) {return {used:null,error:{name:(error as Error).name,message:(error as Error).message}};}
}
function scratchFields(s:ReturnType<typeof createBurnScratch>) {
 return {inputs:s.inputs,acc:s.acc,atmosphere:s.atmosphere,duration:s.duration,capped:s.capped};
}
function compare(pitch:number,booster=false,setup?:(work:UnpoweredFallWork)=>void,stimulus?:(s:ReturnType<typeof createBurnScratch>)=>void) {
 const {state,model}=source(booster);
 const a=createUnpoweredFallWork(state,model.height/2,model,rad(pitch));
 const b=createUnpoweredFallWork(state,model.height/2,model,rad(pitch));
 const sa=createBurnScratch(),sb=createPrototypeScratch();
 if(setup){setup(a);setup(b);}if(stimulus){stimulus(sa);stimulus(sb.burn);}
 const before=cloneState(state);
 const actual=caught(()=>advancePrototypeFall(b,1,sb)),expected=caught(()=>advanceUnpoweredFall(a,1,sa));
 expect(actual).toEqual(expected);expect(checkpoint(b)).toEqual(checkpoint(a));
 expect(scratchFields(sb.burn)).toEqual(scratchFields(sa));expect(state).toEqual(before);
 expect(sb.context.state).toBeNull();expect(sb.context.held).toBeNull();expect(sb.context.model).toBe(SHIP);
}
const bits=new DataView(new ArrayBuffer(8));
function adjacent(value:number,up:boolean) {
 if(value===0)return up?Number.MIN_VALUE:-Number.MIN_VALUE;
 bits.setFloat64(0,value);bits.setBigUint64(0,bits.getBigUint64(0)+(up===(value>0)?1n:-1n));return bits.getFloat64(0);
}

describe('held interface independent exact boundary proofs',()=>{
 it('matches complete force scratch and continuation across dense pitch and control domains',()=>{
  for(const booster of [false,true])for(let index=0;index<=256;index++){
   const angle=-2*Math.PI+index/256*4*Math.PI;
   compare(angle,booster);
   compare(angle,booster,work=>{work.state!.vehicle.frontFinExtension=booster?75:25;work.state!.world.wind=12;work.referenceWind=12;});
  }
 });
 it('matches signed-zero, adjacent quadrant, nonfinite and extreme incidence with full scratch/errors',()=>{
  const edges=[-Math.PI,-Math.PI/2,-0,0,Math.PI/2,Math.PI];
  for(const booster of [false,true])for(const edge of edges)for(const pitch of [adjacent(edge,false),edge,adjacent(edge,true)])
   for(const speed of [0,-0,Number.MIN_VALUE,-Number.MIN_VALUE,-50])compare(pitch,booster,work=>{work.vx=speed;work.vy=speed;});
  for(const booster of [false,true])for(const pitch of [Number.MAX_VALUE,-Number.MAX_VALUE,NaN,Infinity,-Infinity])compare(pitch,booster);
 });
 it('retains all scratch inputs, conditional fixed thrust, nonfinite products and finite cancellation',()=>{
  const values=[0,-0,Number.MIN_VALUE,-Number.MIN_VALUE,1,-1,1e-200,1e200,NaN,Infinity,-Infinity];
  for(const key of ['gimbalPointingDirection','pitch','thrustAcceleration','fixedThrustAcceleration'] as const)
   for(const value of values)for(const fixed of [0,-0,.37])compare(.3,false,undefined,s=>{
    Object.assign(s.inputs,{[key]:value});s.inputs.fixedThrustAcceleration=key==='fixedThrustAcceleration'?value:fixed;
   });
 });
 it('preserves prepared forcing, overflow and inventory exceptions and partial scratch',()=>{
  for(const setup of [
   (w:UnpoweredFallWork)=>{w.vx=NaN;},(w:UnpoweredFallWork)=>{w.vx=Infinity;},
   (w:UnpoweredFallWork)=>{w.vx=-Infinity;},
   (w:UnpoweredFallWork)=>{w.h=0;w.vx=1e154;w.vy=0;},
   (w:UnpoweredFallWork)=>{w.zeroControlArea=w.model.maxArea;w.zeroControlColumnArea=50;w.pitch=rad(NaN);},
   (w:UnpoweredFallWork)=>{w.state!.damage!.components.pop();},
   (w:UnpoweredFallWork)=>{w.state!.damage=null;w.vx=NaN;},
  ])for(const booster of [false,true])compare(.3,booster,setup);
 });
 it('preserves damage roots and actual control loss, including complete errors',()=>{
  for(const booster of [false,true])for(const variant of ['hot','unavailable','invalid-root','invalid-temperature','detached','failed'])
   compare(.3,booster,w=>{for(const c of w.state!.damage!.components){
    if(variant==='hot')c.root.temperature=1100;
    if(variant==='unavailable')c.root.temperature=1200;
    if(variant==='invalid-root'){c.root.valid=false;c.root.temperature=NaN;}
    if(variant==='invalid-temperature')c.root.temperature=NaN;
    if(variant==='detached')c.attached=false;
    if(variant==='failed')c.permanentFailure=ComponentFailure.ProofExceeded;
   }});
 });
 it('pays exactly two ISA/force queries per midpoint, zero for no work, and 8000 at cap',()=>{
  const {state,model}=source();Object.assign(state.kinematics,{altitude:60000,speedX:0,speedY:7000});
  const work=createUnpoweredFallWork(state,model.height/2,model),owned=createPrototypeScratch();
  const spy=vi.spyOn(isa,'isaAtmosphereInto');
  try {
   for(const budget of [-Infinity,-1,0,.9,NaN]){expect(advancePrototypeFall(work,budget,owned)).toBe(0);expect(spy).not.toHaveBeenCalled();expect(owned.context.state).toBeNull();expect(owned.context.held).toBeNull();expect(owned.context.model).toBe(SHIP);}
   expect(advancePrototypeFall(work,1.9,owned)).toBe(1);expect(spy).toHaveBeenCalledTimes(2);
   expect(advancePrototypeFall(work,Infinity,owned)).toBe(3999);expect(spy).toHaveBeenCalledTimes(8000);
   expect(advancePrototypeFall(work,4000,owned)).toBe(0);expect(spy).toHaveBeenCalledTimes(8000);expect(work.steps).toBe(4000);
  } finally {spy.mockRestore();}
 });
 it.each([1,37,512,4000])('matches sliced interleaving with positively effective mutable scratch at budget %s',size=>{
  const sources=[source(),source(true)];
  const pairs=sources.map(({state,model})=>({state,model,before:cloneState(state),a:createUnpoweredFallWork(state,model.height/2,model),b:createUnpoweredFallWork(state,model.height/2,model)}));
  const sa=createBurnScratch(),sb=createPrototypeScratch();let turn=0;
  while(pairs.some(p=>!p.a.done))for(const p of pairs){
   turn++;
   for(const s of [sa,sb.burn]){s.inputs.gimbalPointingDirection=rad(turn%2?.23:-.37);s.inputs.pitch=rad(turn%3?.61:-.19);s.inputs.thrustAcceleration=.5;s.inputs.fixedThrustAcceleration=turn%2?.17:0;}
   expect(caught(()=>advancePrototypeFall(p.b,size,sb))).toEqual(caught(()=>advanceUnpoweredFall(p.a,size,sa)));
   expect(checkpoint(p.b)).toEqual(checkpoint(p.a));expect(scratchFields(sb.burn)).toEqual(scratchFields(sa));
   expect(sb.context.state).toBeNull();expect(sb.context.held).toBeNull();expect(sb.context.model).toBe(SHIP);
  }
  for(const p of pairs){const unmodified=createFallResult();unpoweredFallInto(p.state,p.model.height/2,createBurnScratch(),unmodified,p.model);
   expect(p.a.result).not.toEqual(unmodified);expect(p.state).toEqual(p.before);}
 });
 it('preserves synchronous thrown source/work retention while clearing only added context pointers',()=>{
  const {state,model}=source();state.kinematics.pitch=rad(NaN);

   expect(sb.burn.fallWork.state).toBeNull();expect(sb.context.state).toBeNull();expect(sb.context.held).toBeNull();expect(sb.context.model).toBe(SHIP);
  }
 });

import fs from 'node:fs';import assert from 'node:assert/strict';
const dir=process.argv[2];if(!dir)throw Error('Exact ownership proof receipt required');
const read=n=>JSON.parse(fs.readFileSync(`${dir}/${n}`,'utf8'));
assert.deepEqual(read('before-pins.json'),read('after-pins.json'));
assert.deepEqual(read('runner-closed.json'),{closed:true,unhandledErrors:0,fileErrors:0,nativeExitCode:0});
assert.equal(fs.existsSync(`${dir}/runner-failure.json`),false);
const processResult=read('ownership.process.json');assert.deepEqual(processResult.exit,{code:0,signal:null});assert.equal(processResult.failure,null);assert.equal(processResult.closed,true);assert.deepEqual(processResult.cleanup.executing,[]);
assert.ok(read('independent-owned-absence.json').every(row=>row.executing.length===0));
const report=read('test-results.json');assert.equal(report.numTotalTests,6);assert.equal(report.numPassedTests,6);assert.equal(report.numFailedTests,0);assert.equal(report.numPendingTests,0);assert.equal(report.success,true);assert.equal(report.testResults.length,1);
const tests=report.testResults[0].assertionResults;assert.equal(tests.length,6);
const names=["complete prediction receipt ownership and exact primitives", "each complete input or operation mismatch declines sharing", "unsupported mutable observations never retain aliases", "scenario first pays once and only the later guidance consumer emits original trace files", "pending framework result and identical active mock operations decline sharing", "nonfrozen external model is never borrowed into a stored observation or frozen"];assert.deepEqual(tests.map(t=>t.fullName).sort(),names.sort());
for(const test of tests){assert.equal(test.status,'passed');assert.deepEqual(test.failureMessages,[]);}
fs.writeFileSync(`${dir}/finalized-ownership-proof.json`,JSON.stringify({fakeOwnershipRecipeValidated:true,tests:6,publicLifecycleProof:false,productFlightAcceptance:false},null,2)+'\n',{flag:'wx'});

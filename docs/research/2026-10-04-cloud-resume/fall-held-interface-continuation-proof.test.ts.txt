import {describe,it,expect} from 'vitest';
import {createFallResult,createUnpoweredFallWork,FALL_STEP_CAP} from '$core/control/guidance-physics';
import {advancePrototypeFall as advanceUnpoweredFall,createPrototypeScratch as createBurnScratch,prototypeFallInto as unpoweredFallInto} from './PROTOTYPE_PLACEHOLDER';
import {ALL_SCENARIOS,createScenarioState} from '$core/scenarios';
import {cloneState} from '$core/state';import {SHIP} from '$core/vehicle';import {rad} from '$core/units';
import {originalUnpoweredFall} from '../../../tests/proofs/fixtures/unpowered-fall-original';
function source(h:number,vx:number,vy:number){const s=createScenarioState(ALL_SCENARIOS.find(p=>p.id==='reentry')!);s.kinematics.altitude=h;s.kinematics.speedX=vx;s.kinematics.speedY=vy;return s;}
describe('same midpoint fall continuation',()=>{
 it.each([[79000,2000,-50],[1000,0,-10]])('preserves the original synchronous bits for h=%s', (h,vx,vy)=>{
  const s=source(h,vx,vy),before=cloneState(s),expected=createFallResult();unpoweredFallInto(s,SHIP.height/2,createBurnScratch(),expected,SHIP,rad(0));
  expect(expected).toEqual(originalUnpoweredFall(s,SHIP.height/2,SHIP,rad(0)));
  expect(expected.reached).toBe(true);
  for(const slice of [1,37,512,4000]){
   const work=createUnpoweredFallWork(s,SHIP.height/2,SHIP,rad(0)),scratch=createBurnScratch();let charged=0;
   while(!work.done){const used=advanceUnpoweredFall(work,slice,scratch);expect(used).toBeLessThanOrEqual(slice);charged+=used;}
   expect(work.result).toEqual(expected);expect(charged).toBe(work.steps);expect(s).toEqual(before);
  }
 });
 it('retains exactly4000 iterations and the uncrossed capped result',()=>{
  const s=source(60000,0,7000),scratch=createBurnScratch(),work=createUnpoweredFallWork(s,SHIP.height/2,SHIP,rad(0));let slices=0,charged=0;
  while(!work.done){charged+=advanceUnpoweredFall(work,512,scratch);slices++;}
  expect(charged).toBe(FALL_STEP_CAP);expect(slices).toBe(8);expect(work.result.reached).toBe(false);expect(work.result.time).toBeNaN();expect(work.result.downRange).toBe(0);
  expect(advanceUnpoweredFall(work,512,scratch)).toBe(0);
 });
 it('rejects invalid historical mass without a numerical slice',()=>{
  const s=source(1000,0,-10);s.damage=null;s.vehicle.vehicleMass=0;
  const work=createUnpoweredFallWork(s,SHIP.height/2,SHIP,rad(0));expect(work.done).toBe(true);
  expect(advanceUnpoweredFall(work,512,createBurnScratch())).toBe(0);expect(work.result.reached).toBe(false);expect(work.result.time).toBeNaN();expect(work.result.downRange).toBeNaN();
 });
 it('uses the caller scratch continuation repeatedly without a new per-call record',()=>{
  const s=source(1000,0,-10),scratch=createBurnScratch(),owned=scratch.burn.fallWork,out=createFallResult();
  for(let i=0;i<3;i++){unpoweredFallInto(s,SHIP.height/2,scratch,out,SHIP,rad(0));expect(scratch.burn.fallWork).toBe(owned);expect(out).toEqual(originalUnpoweredFall(s,SHIP.height/2,SHIP,rad(0)));}
 });
 it('retains floor semantics and the mechanical cap for every numeric slice budget',()=>{
  const s=source(60000,0,7000),scratch=createBurnScratch();
  for(const budget of [-Infinity,-1,0,.9,NaN]){
   const work=createUnpoweredFallWork(s,SHIP.height/2,SHIP,rad(0));
   expect(advanceUnpoweredFall(work,budget,scratch)).toBe(0);expect(work.steps).toBe(0);
  }
  const work=createUnpoweredFallWork(s,SHIP.height/2,SHIP,rad(0));
  expect(advanceUnpoweredFall(work,1.9,scratch)).toBe(1);
  expect(advanceUnpoweredFall(work,Infinity,scratch)).toBe(3999);
  expect(work.steps).toBe(4000);expect(work.done).toBe(true);
 });
});

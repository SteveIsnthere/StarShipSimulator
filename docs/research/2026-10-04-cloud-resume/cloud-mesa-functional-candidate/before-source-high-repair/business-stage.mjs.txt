import assert from 'node:assert/strict';
import {readFileSync,writeFileSync,mkdtempSync,chmodSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {pathToFileURL} from 'node:url';
import {resolve} from 'node:path';
const sha=path=>createHash('sha256').update(readFileSync(path)).digest('hex');
const [inputPath,inputSha]=process.argv.slice(2);assert.equal(sha(inputPath),inputSha);const m=JSON.parse(readFileSync(inputPath,'utf8'));
assert.equal(m.kind,'mesa-functional-stage-owner-input-v1');assert.equal(m.purpose,'mesa-functional-webgl2');assert.equal(m.mode,'build');assert.equal(m.outerMs,120000);assert.equal(process.cwd(),m.root);
for(const key of ['node','helper','preparer','builder','qualifier','pins','stage','observer','python','bridge'])assert.equal(sha(m[key].path),m[key].sha,key);
assert.equal(sha(process.execPath),m.node.sha);assert.equal(process.version,'v22.23.3');assert(!process.env.NODE_OPTIONS);
const deadline=Number(process.env.WIDTH2_ROOT_DEADLINE);assert(Number.isFinite(deadline)&&deadline>Date.now()&&deadline<=Date.now()+112000);
const input=JSON.parse(readFileSync(m.qualifier.path,'utf8'));assert.equal(input.receipt,m.setup);assert.equal(input.filePins[m.builder.path],m.builder.sha);
const stage=mkdtempSync('/tmp/starship-width2-stage.');chmodSync(stage,0o700);process.stdout.write(JSON.stringify({stageReceipt:stage})+'\n');
const save=(name,value)=>writeFileSync(resolve(stage,name),JSON.stringify(value,null,2)+'\n',{flag:'wx'});
const {runOwnedCommand,verifyNoOwnedProcesses}=await import(pathToFileURL(m.helper.path).href);
let code=null,error=null,actual=null;try{code=await runOwnedCommand({name:'preparation',args:[m.builder.path,m.qualifier.path,m.qualifier.sha],root:m.root,receipt:stage,env:{...process.env,EGL_PLATFORM:'surfaceless'},deadlineMs:Math.min(104000,deadline-Date.now())});}catch(e){error=e;}
try{actual=JSON.parse(readFileSync(resolve(stage,'preparation.process.json'),'utf8'));}catch(e){error??=e;}
save('actual-child-exit.json',{code,exit:actual?.exit??null,error:error?String(error.stack??error):null});
try{assert.equal(code,0);assert.deepStrictEqual(actual.exit,{code:0,signal:null});assert.equal(actual.failure,null);assert.equal(actual.finished,true);assert.equal(actual.closed,true);assert.equal(actual.cleanup.executing.length,0);
 const outcome=JSON.parse(readFileSync(resolve(m.setup,'outcome.json'),'utf8'));assert.equal(outcome.completed,true);assert.equal(outcome.error,null);assert.equal(outcome.draws,20);assert.deepStrictEqual(outcome.errors,[]);assert.equal(outcome.performanceAcceptance,false);
 const cleanup=JSON.parse(readFileSync(resolve(m.setup,'cleanup.json'),'utf8'));assert.equal(cleanup.verifiedDescendantAbsence,true);assert.equal(cleanup.remaining.length,0);assert.equal(cleanup.inaccessibleOwned.length,0);
 for(const key of ['node','helper','preparer','builder','qualifier','pins','stage','observer','python','bridge'])assert.equal(sha(m[key].path),m[key].sha,key);assert.equal(sha(inputPath),inputSha);assert.equal(sha(process.execPath),m.node.sha);assert(verifyNoOwnedProcesses().every(r=>r.executing.length===0));assert(Date.now()<=deadline);
}catch(e){error??=e;}
save('stage-result.json',{completed:!error,code,error:error?String(error.stack??error):null,actualReceipt:m.setup,acceptance:'Functional Mesa WebGL2 only; ALL root/helper/browser identities and actual result HIGH required'});process.stdout.write(JSON.stringify({stageReceipt:stage,actualReceipt:m.setup,completed:!error})+'\n');if(error)throw error;

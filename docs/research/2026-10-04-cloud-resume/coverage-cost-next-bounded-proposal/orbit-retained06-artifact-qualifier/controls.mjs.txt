import assert from 'node:assert/strict';
import {readFileSync} from 'node:fs';
import {verifyRetainedIdentity} from './policy.mjs';
const originalCertificate = JSON.parse(readFileSync(process.argv[2]));
const make = () => {const certificate = structuredClone(originalCertificate); return {certificate, raw: {kind: 'original-orbit18-uncertified-public-compiler-identities-v1', certified: false, nativeImported: false, paidHistoriesExecuted: false, entry: {path: certificate.expectedGraphs[0].facadeModuleId}, graphs: structuredClone(certificate.expectedGraphs)}, helper: certificate.runtime.helperText};};
const cases = [], positive = make(); assert.equal(verifyRetainedIdentity(positive.raw, positive.certificate, positive.helper).generatedRuntimeCount, 1); cases.push('exact retained metadata accepted');
const deny = (name, message, mutate) => {const fixture = make(); mutate(fixture); assert.throws(() => verifyRetainedIdentity(fixture.raw, fixture.certificate, fixture.helper), error => error instanceof assert.AssertionError && error.code === 'ERR_ASSERTION' && error.message.includes(message)); cases.push(name);};
const graph = (fixture, mutate) => {mutate(fixture.raw.graphs); fixture.certificate.expectedGraphs = structuredClone(fixture.raw.graphs);};
deny('changed raw graph denied', 'Exact retained public graph', f => {f.raw.graphs[1].imports.push('unknown.js');});
deny('changed helper denied', 'Exact generated helper bytes', f => {f.helper += '\n';});
deny('unknown generated ID denied', 'Unknown virtual/query ID', f => graph(f, g => {g[1].moduleIds[0] = '\0unknown/runtime.js';}));
deny('query source denied', 'Unknown virtual/query ID', f => graph(f, g => {g[1].moduleIds[1] += '?unknown';}));
deny('duplicate module denied', 'Duplicate module ID', f => graph(f, g => {g[1].moduleIds.push(g[1].moduleIds[1]);}));
deny('duplicate chunk denied', 'Duplicate chunk', f => graph(f, g => {g[1].fileName = g[0].fileName;}));
deny('wrong helper placement denied', 'Generated helper chunk role', f => graph(f, g => {g[1].moduleIds.shift(); g[0].moduleIds.push(f.certificate.runtime.id);}));
deny('missing helper denied', 'Exact one generated runtime', f => graph(f, g => {g[1].moduleIds.shift();}));
deny('missing original authority denied', 'Missing original authority', f => {delete f.certificate.originalIds[f.raw.graphs[1].moduleIds[1]];});
deny('omitted original module denied', 'Complete original graph', f => graph(f, g => {g[1].moduleIds.pop();}));
deny('wrong entry facade denied', 'Exact entry facade', f => graph(f, g => {g[0].facadeModuleId += '.foreign';}));
deny('foreign import denied', 'Known local import closure', f => graph(f, g => {g[0].imports = ['node:foreign'];}));
deny('dynamic import denied', 'No dynamic closure', f => graph(f, g => {g[0].dynamicImports = ['simulation-BPnlkAel.js'];}));
deny('forged certification denied', 'true', f => {f.raw.certified = true;});
console.log(JSON.stringify({kind: 'original-orbit18-retained-identity-inert-controls-v1', completed: true, cases, actualNativeImports: 0, physicalCalls: 0, artifactScans: 0}));

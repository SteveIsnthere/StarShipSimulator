// UNSELECTED future Node public hook; only after full orbit qualification + policy controls.
import assert from 'node:assert/strict';
import {registerHooks} from 'node:module';
import {readFileSync,appendFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {pathToFileURL,fileURLToPath} from 'node:url';
import {types} from 'node:util';
import {identifyOrbitParent,requireOrbitSourceCertificate,orbitAliasInventory} from './original-parent-policy.mjs';
const[input,expected]=[process.env.NATIVE_ORBIT_INPUT,process.env.NATIVE_ORBIT_INPUT_SHA];assert(input&&expected,'Explicit original orbit qualification required');
const sha=p=>createHash('sha256').update(readFileSync(p)).digest('hex');assert.equal(sha(input),expected);const m=JSON.parse(readFileSync(input));assert.equal(m.kind,'accepted-original-orbit18-native-hook-input-v1');assert.equal(m.original.path,m.root+'/tests/core/orbit.test.ts');assert.equal(m.legacy.path,m.root+'/tests/core/legacy-models.ts');
for(const row of [m.original,m.legacy,m.loader,m.policy,...Object.values(m.facades)])assert.equal(sha(row.path),row.sha);assert.equal(sha(fileURLToPath(import.meta.url)),m.loader.sha);assert.deepEqual(Object.keys(m.facades).sort(),[...orbitAliasInventory].sort());
const certificates=new Map();const record=v=>appendFileSync(m.journal,JSON.stringify(v)+'\n');
function canonical(url,row){if(row===m.original)return identifyOrbitParent(url,row);assert.equal(url,pathToFileURL(row.path).href,'Legacy loaded URL must be exact bare');return{raw:url,canonical:url};}
function certify(url,row,result){const identity=canonical(url,row);assert(['module','module-typescript'].includes(result.format));const source=result.source;assert(!types.isProxy(source),'Proxy returned source denied');assert(typeof source==='string'||Buffer.isBuffer(source)||ArrayBuffer.isView(source),'Public actual TS loaded source required');const bytes=typeof source==='string'?Buffer.from(source):Buffer.from(source.buffer,source.byteOffset,source.byteLength);const loadedSourceSha=createHash('sha256').update(bytes).digest('hex');assert.equal(loadedSourceSha,row.sha);assert.equal(sha(row.path),row.sha);assert(!certificates.has(row.path),'Multiple original/legacy loads denied');const certificate={original:row,loadedSourceSha,format:result.format,identity};certificates.set(row.path,certificate);record({event:'original-source-certified',certificate});}
registerHooks({resolve(specifier,context,nextResolve){try{const parent=context.parentURL;if(specifier==='./legacy-models'){requireOrbitSourceCertificate(certificates.get(m.original.path),m.original,parent);return{url:pathToFileURL(m.legacy.path).href,shortCircuit:true};}if(specifier.startsWith('$core/')){let row=m.original,certificate=certificates.get(row.path);if(parent===pathToFileURL(m.legacy.path).href){row=m.legacy;certificate=certificates.get(row.path);assert.equal(specifier,'$core/constants','Only constants native binding from legacy');canonical(parent,row);assert(certificate&&certificate.loadedSourceSha===row.sha);}else requireOrbitSourceCertificate(certificate,row,parent);assert(orbitAliasInventory.includes(specifier),'Out-of-inventory orbit alias');const facade=m.facades[specifier];assert.equal(sha(facade.path),facade.sha);record({event:'native-orbit-alias',specifier,parent,sourceCertificate:certificate});return{url:pathToFileURL(facade.path).href,shortCircuit:true};}return nextResolve(specifier,context);}catch(error){record({event:'resolve-denied',specifier,parent:context.parentURL,error:String(error)});throw error;}},load(url,context,nextLoad){try{const result=nextLoad(url,context);if(url.startsWith('file:'))for(const row of [m.original,m.legacy])if(fileURLToPath(url.split('?')[0])===row.path)certify(url,row,result);return result;}catch(error){record({event:'load-denied',url,contextFormat:context.format??null,error:String(error)});throw error;}}});

const oracle=graphOracle(left as never,right),DT=1/120,started=performance.now(),histories=[];
const same=(a:unknown,b:unknown,label:string)=>assert(Object.is(a,b),label);

function initial(api:typeof left,altitude:number,speed:number){const s=api.createInitialState(),C=api.orbitConstants;s.kinematics.altitude=altitude;s.kinematics.distanceToPlanetCenter=C.planetRadius+altitude;s.kinematics.speedX=api.groundTangentialSpeed(C.planetRadius+altitude,speed);s.kinematics.speedY=0;s.kinematics.trueSpeed=Math.abs(s.kinematics.speedX);s.kinematics.pitch=Math.PI/2 as never;return s;}
function inertial(api:typeof left,s:ReturnType<typeof initial>){return api.inertialTangentialSpeed(s.kinematics.distanceToPlanetCenter,s.kinematics.speedX);}
function history(name:string,altitude:number,speed:number,steps:number){let a=initial(left,altitude,speed),b=initial(right,altitude,speed),minA=Infinity,minB=Infinity,maxA=-Infinity,maxB=-Infinity,peakStepA=0,peakStepB=0;oracle.compare(a,b,name+':initial');for(let i=0;i<steps;i++){assert(performance.now()-started<=850000,'Research850s supervision; original physical history unchanged');const previousA=a,previousB=b;a=left.step(a,DT);b=right.step(b,DT);oracle.compare(previousA,previousB,name+':previous:'+i);oracle.compare(a,b,name+':returned:'+i);if(a.kinematics.altitude>maxA){maxA=a.kinematics.altitude;peakStepA=i;}if(b.kinematics.altitude>maxB){maxB=b.kinematics.altitude;peakStepB=i;}minA=Math.min(minA,a.kinematics.altitude);minB=Math.min(minB,b.kinematics.altitude);same(minA,minB,name+':min');same(maxA,maxB,name+':max');same(peakStepA,peakStepB,name+':peakStep');same(inertial(left,a),inertial(right,b),name+':inertial');}histories.push({name,steps,min:minA,max:maxA,peakStep:peakStepA});return{a,b,min:minA,max:maxA,peakStep:peakStepA};}

const energy=history('original ten minute energy',300000,v300,72000);const e0=left.specificOrbitalEnergy(r300,v300);const e1=left.specificOrbitalEnergy(energy.a.kinematics.distanceToPlanetCenter,Math.hypot(inertial(left,energy.a),energy.a.kinematics.speedY));same(e1,right.specificOrbitalEnergy(energy.b.kinematics.distanceToPlanetCenter,Math.hypot(inertial(right,energy.b),energy.b.kinematics.speedY)),'energy outcome');assert(Math.abs((e1-e0)/e0)<1e-3);

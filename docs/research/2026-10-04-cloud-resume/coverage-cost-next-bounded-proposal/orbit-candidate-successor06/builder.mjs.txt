import assert from 'node:assert/strict';
import {build,resolveConfig} from 'vite';
import {readFileSync,writeFileSync,readdirSync} from 'node:fs';
import {resolve,dirname,isAbsolute} from 'node:path';
import {createHash} from 'node:crypto';
const [input,expected]=process.argv.slice(2),sha=p=>createHash('sha256').update(readFileSync(p)).digest('hex');assert.equal(sha(input),expected);const m=JSON.parse(readFileSync(input));assert.equal(m.kind,'original-orbit18-qualification-input-v1');
for(const[p,h]of Object.entries(m.files))assert.equal(sha(p),h,p);
const config=await resolveConfig({configFile:resolve(m.root,'vite.config.ts')},'build','production');
writeFileSync(resolve(m.receipt,'production-options.json'),JSON.stringify({target:config.build.target,minify:config.build.minify,sourcemap:config.build.sourcemap,cssMinify:config.build.cssMinify,base:config.base,mode:config.mode,adaptation:'ESM library entry/output only; no numerical transform or passive counters'},null,2)+'\n',{flag:'wx'});
assert.equal(sha(m.entry.path),m.entry.sha);
const built=await build({configFile:resolve(m.root,'vite.config.ts'),build:{target:config.build.target,minify:config.build.minify,sourcemap:config.build.sourcemap,cssMinify:config.build.cssMinify,outDir:resolve(m.receipt,'plain'),emptyOutDir:true,lib:{entry:m.entry.path,formats:['es'],fileName:()=> 'entry.mjs'}}});
// Preserve bounded public compiler identity metadata BEFORE strict certification.
const outputs=Array.isArray(built)?built:[built];assert(outputs.length<=64,'Compiler output bound');const rawGraphs=[];let rawModules=0;const text=(value,label)=>{assert.equal(typeof value,'string',label);assert(value.length<=8192,label+' string bound');return value;};const list=(values,label)=>{assert(Array.isArray(values)&&values.length<=4096,label+' array bound');return values.map(value=>text(value,label));};
for(const output of outputs){assert(output&&Array.isArray(output.output),'Public build output required');for(const chunk of output.output){if(chunk.type!=='chunk')continue;assert(rawGraphs.length<128,'Compiler chunk bound');const ids=list(chunk.moduleIds,'module ID');rawModules+=ids.length;assert(rawModules<=65536,'Compiler module aggregate bound');rawGraphs.push({fileName:text(chunk.fileName,'chunk file'),facadeModuleId:chunk.facadeModuleId===null?null:text(chunk.facadeModuleId,'facade ID'),moduleIds:ids,imports:list(chunk.imports,'import'),dynamicImports:list(chunk.dynamicImports,'dynamic import')});}}
const rawGraphText=JSON.stringify({kind:'original-orbit18-uncertified-public-compiler-identities-v1',certified:false,nativeImported:false,paidHistoriesExecuted:false,entry:m.entry,originalTest:m.originalTest,graphs:rawGraphs},null,2)+'\n';assert(Buffer.byteLength(rawGraphText)<=2*1024*1024,'Compiler identity evidence2MiB bound');writeFileSync(resolve(m.receipt,'uncertified-compiler-identities.json'),rawGraphText,{flag:'wx'});
const files=[],mapped=new Set(),moduleGraphs=[];for(const output of outputs){assert(output&&Array.isArray(output.output),'Public build output required');for(const chunk of output.output){if(chunk.type!=='chunk')continue;const moduleIds=chunk.moduleIds.map(id=>{assert(!id.includes(String.fromCharCode(0))&&!id.includes('?'),'Uncertified virtual/query module ID');assert.equal(m.files[id],sha(id),'Original module ID/source authority');return{id,sha:sha(id)};});moduleGraphs.push({fileName:chunk.fileName,facadeModuleId:chunk.facadeModuleId,moduleIds,imports:chunk.imports,dynamicImports:chunk.dynamicImports});}}
for(const name of readdirSync(resolve(m.receipt,'plain')).sort()){const path=resolve(m.receipt,'plain',name);files.push({path,sha:sha(path)});if(!name.endsWith('.map'))continue;const map=JSON.parse(readFileSync(path));assert.equal(map.sources.length,map.sourcesContent.length);for(let i=0;i<map.sources.length;i++){assert.equal(typeof map.sourcesContent[i],'string');assert(!map.sources[i].includes('://'));const source=isAbsolute(map.sources[i])?map.sources[i]:resolve(dirname(path),map.sourceRoot??'',map.sources[i]);assert.equal(map.sourcesContent[i],readFileSync(source,'utf8'));assert.equal(m.files[source],sha(source),'Every mapped source must have external source authority');mapped.add(source);}}
for(const source of ['src/core/state.ts','src/core/step.ts','src/core/physics/gravity.ts','src/core/constants.ts','tests/core/legacy-models.ts'])assert(mapped.has(resolve(m.root,source)),'Required source absent '+source);
writeFileSync(resolve(m.receipt,'compiled-manifest.json'),JSON.stringify({kind:'original-orbit18-compiled-artifacts-v1',files,mapped:[...mapped].sort(),moduleGraphs,entrySource:m.entry,originalOrbit:m.originalTest,productionOptions:{path:resolve(m.receipt,'production-options.json'),sha:sha(resolve(m.receipt,'production-options.json'))}},null,2)+'\n',{flag:'wx'});
for(const[p,h]of Object.entries(m.files))assert.equal(sha(p),h,p);assert.equal(sha(input),expected);

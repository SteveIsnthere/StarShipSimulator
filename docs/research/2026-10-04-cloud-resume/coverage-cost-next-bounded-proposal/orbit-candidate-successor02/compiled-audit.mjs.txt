import assert from 'node:assert/strict';
import {readFileSync,readdirSync,lstatSync,realpathSync} from 'node:fs';
import {resolve,dirname,isAbsolute} from 'node:path';
import {createHash} from 'node:crypto';
const sha=p=>createHash('sha256').update(readFileSync(p)).digest('hex');
export function verifyCompiledArtifacts(m,accepted){
 assert.equal(accepted.kind,'independently-accepted-original-orbit18-production-artifacts-v1');assert.equal(accepted.compiledArtifactsAccepted,true);assert.equal(accepted.historyExecutionAccepted,false);assert.deepEqual(accepted.originalTest,m.originalTest);assert.deepEqual(accepted.sourceAuthority,m.authorityRole);for(const key of ['manifest','productionOptions','producerResult','independentReview','independentReviewPins']){const row=accepted[key];assert.equal(m.files[row.path],row.sha,'External accepted compiled lineage pin');assert.equal(sha(row.path),row.sha);}
 const manifest=JSON.parse(readFileSync(accepted.manifest.path));assert.equal(manifest.kind,'original-orbit18-compiled-artifacts-v1');assert.deepEqual(manifest.originalOrbit,m.originalTest);assert.deepEqual(manifest.entrySource,m.entry);assert.deepEqual(manifest.productionOptions,accepted.productionOptions);
 const directory=resolve(m.receipt,'plain');assert.equal(realpathSync(directory),directory);const actual=readdirSync(directory).sort(),expected=manifest.files.map(r=>{assert.equal(dirname(r.path),directory);assert(!lstatSync(r.path).isSymbolicLink()&&lstatSync(r.path).isFile());assert.equal(sha(r.path),r.sha);return r.path.slice(directory.length+1);}).sort();assert.deepEqual(actual,expected);assert(expected.includes('entry.mjs'));assert(expected.some(s=>s.endsWith('.map')));assert.equal(new Set(expected).size,expected.length);
 const chunks=manifest.moduleGraphs;assert(chunks.length>0);assert.deepEqual(chunks.map(c=>c.fileName).sort(),expected.filter(name=>name.endsWith('.js')||name.endsWith('.mjs')).sort(),'Every executing JS output has its public chunk graph');assert.equal(new Set(chunks.map(c=>c.fileName)).size,chunks.length);assert(chunks.some(c=>c.fileName==='entry.mjs'&&c.facadeModuleId===m.entry.path),'Exact entry facade/module identity');const sourceIds=new Map();for(const c of chunks){assert(expected.includes(c.fileName));if(c.facadeModuleId!==null){assert.equal(m.files[c.facadeModuleId],sha(c.facadeModuleId));}for(const row of c.moduleIds){assert.equal(m.files[row.id],row.sha,'Every original module expected');assert.equal(sha(row.id),row.sha);if(sourceIds.has(row.id))assert.equal(sourceIds.get(row.id),row.sha);sourceIds.set(row.id,row.sha);}for(const name of [...c.imports,...c.dynamicImports]){assert.equal(typeof name,'string');assert(!name.startsWith('/')&&!name.includes('://')&&!name.startsWith('node:'));const target=resolve(directory,name);assert.equal(dirname(target),directory,'Only known local emitted imports');assert(expected.includes(target.slice(directory.length+1)),'Uncertified emitted chunk import');}}
 const mapped=new Set();for(const file of manifest.files.filter(r=>r.path.endsWith('.map'))){const map=JSON.parse(readFileSync(file.path));assert.equal(map.sources.length,map.sourcesContent.length);for(let i=0;i<map.sources.length;i++){const path=isAbsolute(map.sources[i])?map.sources[i]:resolve(dirname(file.path),map.sourceRoot??'',map.sources[i]);assert.equal(typeof map.sourcesContent[i],'string');assert.equal(readFileSync(path,'utf8'),map.sourcesContent[i]);assert.equal(m.files[path],sha(path));mapped.add(path);}}assert.deepEqual([...mapped].sort(),manifest.mapped);for(const path of ['src/core/state.ts','src/core/step.ts','src/core/physics/gravity.ts','src/core/constants.ts','tests/core/legacy-models.ts'])assert(mapped.has(resolve(m.root,path)));
 return{matched:true,files:expected,moduleCount:sourceIds.size,sourceCount:mapped.size,acceptedManifest:accepted.manifest};
}

  assert.equal(typeof v,'object','Unsupported function/symbol/bigint');assert(!seen.has(v),'Repeated identity/cycle not representable');seen.add(v);

}

 function visit(row,depth){assert(Array.isArray(row)&&depth<=64&&++count<=1000000);const[k,v]=row;

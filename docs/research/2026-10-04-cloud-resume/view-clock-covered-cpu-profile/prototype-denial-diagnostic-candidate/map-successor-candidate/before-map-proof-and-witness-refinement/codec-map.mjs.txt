const IntrinsicMap=Map, mapPrototype=Map.prototype, mapForEach=Map.prototype.forEach, mapSet=Map.prototype.set, mapHas=Map.prototype.has;
// Tagged OWN DATA plus exact canonical native Map internal entries only. Never invokes a getter or toJSON; no generic prototype admission.
export function encodeCoverageOwnData(value,approvedFileCoveragePrototype,onPrototypeDenial,onMapWitness){
 const seen=new WeakSet();let count=0;
 function visit(v,depth,path){assert(depth<=64&&++count<=1000000);if(v===undefined)return['undefined'];if(v===null)return['null'];
  if(typeof v==='number')return['number',Object.is(v,-0)?'-0':Number.isFinite(v)?v:String(v)];if(typeof v==='string'||typeof v==='boolean')return[typeof v,v];
  assert.equal(typeof v,'object','Unsupported function/symbol/bigint');assert(!types.isProxy(v),'Proxy provider object unsupported');assert(!seen.has(v),'Repeated identity/cycle not representable');seen.add(v);
  const proto=Object.getPrototypeOf(v),nativeMap=types.isMap(v)&&proto===mapPrototype,kind=nativeMap?'map':Array.isArray(v)?'array':proto===null?'null-object':proto===Object.prototype?'object':proto===approvedFileCoveragePrototype&&approvedFileCoveragePrototype?'file-coverage':null;if(!kind&&onPrototypeDenial)onPrototypeDenial(v,path,approvedFileCoveragePrototype);assert(kind,'Unsupported provider prototype');if(kind==='array'){if(proto!==Array.prototype&&onPrototypeDenial)onPrototypeDenial(v,path,approvedFileCoveragePrototype);assert.equal(proto,Array.prototype);}
  let internal=null;if(kind==='map'){assert.equal(typeof onMapWitness,'function','Exact intrinsic Map witness required');onMapWitness({path,nativeBrand:true,exactMapPrototype:true,prototypeConstructorOwnData:Object.getOwnPropertyDescriptor(proto,'constructor')?.value===IntrinsicMap});internal=[];Reflect.apply(mapForEach,v,[(value,key)=>{assert(internal.length<1000000,'Map entry cap');internal.push([visit(key,depth+1,[...path,'map-key',String(internal.length)]),visit(value,depth+1,[...path,'map-value',String(internal.length)])]);}]);}
  const entries=Reflect.ownKeys(v).map(k=>{assert.equal(typeof k,'string','Symbol own key unsupported');const d=Object.getOwnPropertyDescriptor(v,k);assert('value'in d,'Accessor property unsupported');return[k,visit(d.value,depth+1,[...path,k]),d.writable,d.enumerable,d.configurable];});return kind==='map'?[kind,internal,entries]:[kind,entries];
 }return{encoding:'coverage-own-data-map-v3',value:visit(value,0,[])};
}
export function decodeCoverageOwnData(document,approvedFileCoveragePrototype){
 assert.deepEqual(Object.keys(document).sort(),['encoding','value']);assert.equal(document.encoding,'coverage-own-data-map-v3');let count=0;
 function visit(row,depth){assert(Array.isArray(row)&&depth<=64&&++count<=1000000);const[k,v]=row;let mapOut=null;if(k==='map'){assert.equal(row.length,3);assert(Array.isArray(v));mapOut=new IntrinsicMap();for(const pair of v){assert(Array.isArray(pair)&&pair.length===2);const key=visit(pair[0],depth+1),value=visit(pair[1],depth+1);assert(!Reflect.apply(mapHas,mapOut,[key]),'Duplicate native Map key');Reflect.apply(mapSet,mapOut,[key,value]);}}
  if(k==='undefined'||k==='null'){assert.equal(row.length,1);return k==='undefined'?undefined:null;}if(k!=='map')assert.equal(row.length,2);
  if(k==='number'){if(typeof v==='number'){assert(Number.isFinite(v));return v;}assert(['-0','NaN','Infinity','-Infinity'].includes(v));return v==='-0'?-0:v==='NaN'?NaN:v==='Infinity'?Infinity:-Infinity;}
  if(k==='string'||k==='boolean'){assert.equal(typeof v,k);return v;}
  assert(['array','null-object','object','file-coverage','map'].includes(k));if(k==='file-coverage')assert(approvedFileCoveragePrototype,'Unadmitted FileCoverage prototype');const out=mapOut??(k==='array'?[]:Object.create(k==='null-object'?null:k==='file-coverage'?approvedFileCoveragePrototype:Object.prototype));const properties=k==='map'?row[2]:v;assert(Array.isArray(properties));const keys=new Set();for(const item of properties){assert(Array.isArray(item)&&item.length===5);const[name,value,writable,enumerable,configurable]=item;assert.equal(typeof name,'string');assert(!keys.has(name));keys.add(name);for(const flag of[writable,enumerable,configurable])assert.equal(typeof flag,'boolean');Object.defineProperty(out,name,{value:visit(value,depth+1),writable,enumerable,configurable});}return out;
 }return visit(document.value,0);
}

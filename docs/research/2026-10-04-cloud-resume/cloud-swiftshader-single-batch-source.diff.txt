+import { RendererType, type Renderer, type WebGLRenderer } from 'pixi.js';
+
+/** Narrow batching policy; actual hardware limits and visual sampling stay intact. */
+export function applySwiftShaderBatchPolicy(renderer: Renderer): void {
+  if (renderer.type !== RendererType.WEBGL) return;
+  const webgl = renderer as WebGLRenderer;
+  const extension = webgl.gl.getExtension('WEBGL_debug_renderer_info');
+  if (!extension) return;
+  const observed: unknown = webgl.gl.getParameter(extension.UNMASKED_RENDERER_WEBGL);
+  if (typeof observed !== 'string' || !/\bSwiftShader\b/i.test(observed)) return;
+  const existing = webgl.limits.maxBatchableTextures;
+  if (!Number.isInteger(existing) || existing < 1) return;
+  webgl.limits.maxBatchableTextures = 1;
+}

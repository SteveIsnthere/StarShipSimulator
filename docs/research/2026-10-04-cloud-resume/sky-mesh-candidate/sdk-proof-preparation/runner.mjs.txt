import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {resolve} from 'node:path';
import {startVitest} from 'vitest/node';
const directory=process.env.SKY_SDK_SETUP;
assert(directory && /^\/tmp\/starship-sky-sdk\.[a-f0-9]{32}$/.test(directory));
assert.equal(process.env.STARSHIP_CPU_SLOT_GRANTED,'sky-mesh-public-sdk-proof');
const expected=JSON.parse(readFileSync(resolve(directory,'expected-names.json'),'utf8'));
let ctx,primary;
try {ctx=await startVitest('test',[],{config:resolve(directory,'config.ts'),watch:false,run:true});assert(ctx);}
catch(error){primary=error;}
finally {try {if(ctx)await ctx.close();}catch(error){primary??=error;}}
try {
 if(primary)throw primary;
 assert.equal(process.exitCode??0,0);
 assert.deepEqual(ctx.state.getUnhandledErrors(),[]);
 const files=ctx.state.getFiles();assert.equal(files.length,1);
 const names=[];
 function visit(task) {
  assert.equal(task.mode,'run');assert.notEqual(task.fails,true);
  assert.equal(task.result?.errors?.length??0,0);assert.equal(task.result?.state,'pass');
  if(task.type==='test') {names.push(task.name);assert.equal(task.retry??0,0);assert.equal(task.repeats??0,0);
   assert.equal(task.result?.retryCount??0,0);assert.equal(task.result?.repeatCount??0,0);}
  for(const child of task.tasks??[])visit(child);
 }
 for(const file of files)visit(file);
 assert.deepEqual(names,expected);
 const report=JSON.parse(readFileSync(resolve(directory,'test-results.json'),'utf8'));
 assert.equal(report.numTotalTests,8);assert.equal(report.numPassedTests,8);
 for(const key of ['numFailedTests','numPendingTests','numTodoTests'])assert.equal(report[key],0);
 assert.equal(report.success,true);
 writeFileSync(resolve(directory,'runner-closed.json'),JSON.stringify({closed:true,
  names,unhandledErrors:0,fileErrors:0,nativeExitCode:0,scope:'public SDK CPU only'})+'\n',{flag:'wx'});
} catch(error) {writeFileSync(resolve(directory,'runner-failure.json'),JSON.stringify({error:String(error.stack??error)})+'\n',{flag:'wx'});throw error;}

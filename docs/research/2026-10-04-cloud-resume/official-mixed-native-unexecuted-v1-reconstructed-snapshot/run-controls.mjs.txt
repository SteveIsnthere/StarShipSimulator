import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { digest, parse, inputsEqual } from './integrity.mjs';
import { admit } from './admission.mjs';
const started = Date.now(), deadline = started + 30000;
const receipt = dirname(fileURLToPath(import.meta.url));
const root = process.cwd();
const { approval, manifest } = await admit(process.env.NATIVE_FLIGHT_APPROVAL, process.env.NATIVE_FLIGHT_APPROVAL_SHA);
assert.equal(manifest.receipt, receipt);
const ownerPath = resolve(root, 'scripts/bench/owned-command.mjs');
assert.equal(digest(ownerPath), approval.ownedCommandSha);
const { runOwnedCommand, verifyNoOwnedProcesses } = await import(pathToFileURL(ownerPath).href);
const bound = deadline - Date.now(); assert(bound > 0);
let processReceipt, failure;
try {
  const code = await runOwnedCommand({ name: 'control-process',
    args: [resolve(root, 'node_modules/vitest/vitest.mjs'), 'run', '--config', resolve(receipt, 'control-config.mjs')],
    root, env: { ...process.env }, receipt, deadlineMs: bound });
  processReceipt = parse(resolve(receipt, 'control-process.process.json'));
  assert.equal(code, 0); assert.equal(processReceipt.exit.code, 0);
  assert.equal(processReceipt.exit.signal, null); assert.equal(processReceipt.failure, null);
  assert.equal(processReceipt.closed, true); assert.equal(processReceipt.cleanup.executing.length, 0);
  assert.equal(readFileSync(resolve(receipt, 'control-process.stderr.txt'), 'utf8'), '', 'Unexpected stderr is not allowed');
  const report = parse(resolve(receipt, 'control-tests.json'));
  assert.equal(report.success, true); assert.equal(report.numPassedTests, 3);
  assert.equal(report.numFailedTests, 0);
  assert.equal(report.numTotalTests, 3);
  for (const suite of report.testResults) for (const item of suite.assertionResults) {
    assert.equal(item.status, 'passed'); assert.deepEqual(item.failureMessages, []);
  }
  if (report.unhandledErrors !== undefined) assert.deepEqual(report.unhandledErrors, []);
  const a = parse(resolve(receipt, 'native-a-result.json'));
  const b = parse(resolve(receipt, 'native-b-result.json'));
  const ssr = parse(resolve(receipt, 'ssr-result.json'));
  assert.notEqual(a.pid, b.pid); assert.match(a.starttime, /^\d+$/); assert.match(b.starttime, /^\d+$/);
  assert.equal(a.marker, 'a'); assert.equal(b.marker, 'b');
  assert.deepEqual(a.value, b.value); assert.deepEqual(a.value, ssr.value);
  for (const row of [a, b]) assert(processReceipt.cleanup.tracked.some(owned =>
    owned.pid === row.pid && String(owned.start) === row.starttime));
  writeFileSync(resolve(receipt, 'control-result.json'), JSON.stringify({
    completed: true, originalTwentyOneTestsPassed: false, coverageQualified: false,
    fixtures: [a, b, ssr], approvalSha: process.env.NATIVE_FLIGHT_APPROVAL_SHA }) + '\n', { flag: 'wx' });
} catch (error) { failure = error; }
try {
  await admit(process.env.NATIVE_FLIGHT_APPROVAL, process.env.NATIVE_FLIGHT_APPROVAL_SHA);
  inputsEqual(manifest.materializedInputs);
  assert.equal(digest(process.env.NATIVE_FLIGHT_APPROVAL), process.env.NATIVE_FLIGHT_APPROVAL_SHA);
  assert.equal(digest(approval.manifestPath), approval.manifestSha);
  assert.equal(digest(ownerPath), approval.ownedCommandSha);
  assert.equal(digest(process.execPath), manifest.nodeBinary);
  const absence = verifyNoOwnedProcesses();
  assert(absence.every(row => row.executing.length === 0));
  assert(Date.now() <= deadline, 'Whole admission workload exceeded 30 seconds');
} catch (error) { failure ??= error; }
writeFileSync(resolve(receipt, 'control-owner-result.json'), JSON.stringify({
  completed: !failure, code: failure ? 1 : 0, error: failure ? String(failure.stack ?? failure) : null,
  elapsedMs: Date.now() - started, processReceipt,
  acceptance: 'small controls only; parent independent absence/source checks required' }) + '\n', { flag: 'wx' });
if (failure) throw failure;

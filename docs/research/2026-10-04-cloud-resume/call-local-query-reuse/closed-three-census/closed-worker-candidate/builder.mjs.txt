// Production-option private full-entry build only; no simulation kernel imports.
import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {resolve,relative,dirname} from 'node:path';
import {createHash} from 'node:crypto';
import {pathToFileURL} from 'node:url';
import {admitSource,certifyGeneratedRuntime} from './admission.mjs';
const[input,inputSha]=process.argv.slice(2);const{m,loader}=await admitSource(input,inputSha);
assert.deepEqual(process.execArgv,[]);
const sha=path=>createHash('sha256').update(readFileSync(path)).digest('hex');
const save=(name,value)=>writeFileSync(resolve(m.setup,name),JSON.stringify(value,null,2)+'\n',{flag:'wx'});
const{build,resolveConfig}=await import(pathToFileURL(resolve(m.root,'node_modules/vite/dist/node/index.js')).href);
const configFile=resolve(m.root,'vite.config.ts'),resolved=await resolveConfig({configFile},'build','production');
const options=JSON.parse(readFileSync(resolve(m.root,'scripts/bench/qualification.json'),'utf8')).options;
assert.deepEqual({target:resolved.build.target,minify:resolved.build.minify,sourcemap:resolved.build.sourcemap,cssMinify:resolved.build.cssMinify,mode:resolved.mode,base:resolved.base},Object.fromEntries(['target','minify','sourcemap','cssMinify','mode','base'].map(key=>[key,options[key]])));
assert.deepEqual(Object.keys(m.virtualSources).sort(),['src/core/physics/damage-controls.ts','src/core/physics/damage-thermal.ts']);
assert.equal(sha(m.entry.path),m.entry.sha256);const overrides={};for(const[file,row]of Object.entries(m.virtualSources)){assert.equal(sha(row.path),row.sha256);overrides[resolve(m.root,file)]=readFileSync(row.path,'utf8');}
const directory=resolve(m.setup,'native');
let primary=null;const postErrors=[];
try{
 const built=await build({configFile,plugins:[{name:'closed-three-all-original-paid-census-copy',enforce:'pre',load(id){return Object.hasOwn(overrides,id)?overrides[id]:null;}}],
 build:{target:resolved.build.target,minify:resolved.build.minify,sourcemap:resolved.build.sourcemap,cssMinify:resolved.build.cssMinify,copyPublicDir:false,outDir:directory,emptyOutDir:true,lib:{entry:m.entry.path,formats:['es'],fileName:()=> 'entry.mjs'}}});
 const files=Object.fromEntries(loader.filesUnder(directory).map(file=>[relative(directory,file),sha(file)]));const outputs=Array.isArray(built)?built:[built];
 const chunks=outputs.flatMap(output=>{assert.ok(Array.isArray(output.output));return output.output.filter(item=>item.type==='chunk');});
 const edges=chunks.map(chunk=>({fileName:chunk.fileName,isEntry:chunk.isEntry,facadeModuleId:chunk.facadeModuleId,moduleIds:chunk.moduleIds,imports:chunk.imports,dynamicImports:chunk.dynamicImports,sha256:typeof chunk.code==='string'?createHash('sha256').update(chunk.code).digest('hex'):null}));
 save('full-entry-preliminary-compiler.json',{files,edges});
 const maps={};for(const file of loader.filesUnder(directory).filter(file=>file.endsWith('.map'))){const map=JSON.parse(readFileSync(file,'utf8'));maps[relative(directory,file)]=map;}save('full-entry-preliminary-maps.json',maps);
 assert.ok(chunks.length>0&&chunks.length<=16);const owned=new Set(chunks.map(chunk=>chunk.fileName)),ids=new Set();assert.equal(owned.size,chunks.length);
 for(const edge of edges){assert.equal(edge.sha256,files[edge.fileName]);assert.deepEqual(edge.dynamicImports,[]);for(const imported of edge.imports)assert.ok(owned.has(imported));for(const id of edge.moduleIds)ids.add(id);}
 const generatedRuntime=certifyGeneratedRuntime(m,edges,directory);save('full-entry-generated-runtime-observation.json',generatedRuntime);
 assert.deepEqual([...ids].sort(),[...Object.keys(m.graph).map(file=>resolve(m.root,file)),m.entry.path,'\0rolldown/runtime.js'].sort());
 const entry=edges.find(edge=>edge.fileName==='entry.mjs');assert.equal(entry.isEntry,true);assert.equal(entry.facadeModuleId,m.entry.path);
 const mapped={};for(const[file,map]of Object.entries(maps)){assert.equal(map.version,3);assert.equal(map.sources.length,map.sourcesContent.length);assert.equal(map.sourceRoot??'','');
  const code=resolve(dirname(resolve(directory,file)),map.file);assert.ok(Object.hasOwn(files,relative(directory,code)));
  for(let i=0;i<map.sources.length;i++){const source=resolve(dirname(resolve(directory,file)),map.sources[i]),relativeName=relative(m.root,source);const virtual=m.virtualSources[relativeName];const text=virtual?readFileSync(virtual.path,'utf8'):readFileSync(source,'utf8');assert.equal(map.sourcesContent[i],text);mapped[source]=sha(virtual?.path??source);}}
 const expected={...Object.fromEntries(Object.entries(m.graph).map(([file,h])=>[resolve(m.root,file),m.virtualSources[file]?.sha256??h])),[m.entry.path]:m.entry.sha256};assert.deepEqual(mapped,expected);
 const constants=readFileSync(resolve(m.root,'scripts/test/constants.mjs'));assert.equal(constants.toString(),"export { starBaseXPos, heatLimit } from './entry.mjs';\n");writeFileSync(resolve(directory,'constants.mjs'),constants,{flag:'wx'});files['constants.mjs']=sha(resolve(directory,'constants.mjs'));
 assert.ok(Object.keys(files).length<=32);const bytes=Object.keys(files).reduce((sum,file)=>sum+readFileSync(resolve(directory,file)).length,0);assert.ok(bytes<=64*1024*1024);
 const nonEntry=chunks.filter(chunk=>!chunk.isEntry);assert.equal(nonEntry.length,1);
 const artifact={kind:'closed-three-original-query-census-artifact-v1',directory,receipt:m.setup,files,edges,generatedRuntime,mappedCode:nonEntry[0].fileName,graph:m.graph,virtualSources:m.virtualSources,entrySha256:m.entry.sha256,sourceInput:{path:input,sha256:inputSha},sourceInputsSha256:m.sourceInputs.sha256,bodyApprovalSha256:m.bodyApproval.sha256,freshBuildCompleted:true,allOriginalQueriesPaid:true,queryResultsNeverSubstituted:true,publicSelection:false};
 writeFileSync(resolve(m.setup,'native-artifact.json'),JSON.stringify(artifact,null,2)+'\n',{flag:'wx'});
}catch(error){primary=String(error.stack??error);}
try{await admitSource(input,inputSha);}catch(error){postErrors.push(String(error.stack??error));}
save('full-entry-build-result.json',{completed:primary===null&&postErrors.length===0,primary,postErrors,physicsExecuted:false});if(primary||postErrors.length)process.exitCode=1;

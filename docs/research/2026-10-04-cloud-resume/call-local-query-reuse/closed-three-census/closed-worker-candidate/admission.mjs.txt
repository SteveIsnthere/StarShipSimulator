// Private closed-three census admission. No production/public dispatcher.
import assert from 'node:assert/strict';
import {readFileSync,realpathSync} from 'node:fs';
import {resolve,relative,isAbsolute,dirname} from 'node:path';
import {pathToFileURL} from 'node:url';
import {createHash} from 'node:crypto';
const sha=path=>createHash('sha256').update(readFileSync(path)).digest('hex');
const parse=path=>JSON.parse(readFileSync(path,'utf8'));
function pin(row){assert.equal(typeof row.path,'string');assert.ok(isAbsolute(row.path));assert.match(row.sha256,/^[a-f0-9]{64}$/);assert.equal(sha(row.path),row.sha256);return parse(row.path);}
export async function admitSource(path,expected){
 assert.equal(sha(path),expected);const m=parse(path);assert.equal(m.kind,'closed-three-census-source-input-v1');
 assert.equal(process.version,'v22.23.3');assert.equal(process.platform,'linux');assert.equal(process.arch,'x64');assert.equal(sha(process.execPath),m.nodeSha256);
 assert.equal(process.env.NODE_OPTIONS??'','');assert.equal(process.env.NODE_PATH??'','');assert.equal(process.env.STARSHIP_CPU_SLOT_GRANTED,'query-census-closed-three');
 assert.equal(m.root,realpathSync(m.root));assert.match(relative(m.root,m.setup),/^node_modules\/\.cache\/starship-closed-three-census-[a-f0-9]{32}$/);assert.equal(realpathSync(m.setup),m.setup);
 for(const [file,digest]of Object.entries(m.selected)){assert.ok(isAbsolute(file));assert.equal(sha(file),digest,file);}
 const approval=pin(m.bodyApproval);assert.equal(approval.kind,'closed-three-native-body-approval-v1');assert.equal(approval.status,'approved');assert.equal(approval.publicDispatcher,false);
 assert.equal(approval.generatedRuntimeSha256,m.generatedRuntime.sha256);const generated=pin(m.generatedRuntime);assert.equal(generated.kind,'closed-three-exact-generated-runtime-class-v1');assert.equal(generated.compilerModuleId,'\0rolldown/runtime.js');assert.equal(generated.exactlyOneGeneratedModule,true);assert.equal(generated.mappedSourceCredit,false);for(const[file,h]of Object.entries(generated.producerFiles)){assert.equal(m.selected[file],h);assert.equal(sha(file),h);}assert.equal(sha(generated.retainedProjection.path),generated.retainedProjection.sha256);assert.equal(sha(generated.retainedChunk.path),generated.retainedChunk.sha256);
 assert.equal(approval.sourceInputsSha256,m.sourceInputs.sha256);assert.equal(sha(m.sourceInputs.path),m.sourceInputs.sha256);
 const approvedSourceFiles=parse(m.sourceInputs.path).files;
 for(const name of ['admission.mjs','loader.mjs','config.ts','builder.mjs','entry.ts','observer.mjs']){
  const file=resolve(m.setup,name),row=m.materialized[file];assert.ok(row,'Missing exact private role');
  const original=resolve(m.root,'docs/research/2026-10-04-cloud-resume/call-local-query-reuse/closed-three-census/closed-worker-candidate',name+'.txt');
  assert.equal(row.source,original);assert.equal(row.sourceSha256,approvedSourceFiles[original]);assert.equal(row.sha256,row.sourceSha256);assert.equal(m.selected[file],row.sha256);assert.equal(sha(file),row.sha256);assert.equal(sha(original),row.sourceSha256);
 }
 assert.equal(m.observer.path,resolve(m.setup,'observer.mjs'));assert.equal(m.observer.sha256,m.materialized[m.observer.path].sha256);

 assert.equal(approval.authoritySha256,m.authority.sha256);assert.equal(approval.censusSourceControlsSha256,m.censusControls.sha256);assert.equal(approval.reportingControlsSha256,m.reportingControls.sha256);
 const numerical=pin(m.censusControls);assert.equal(numerical.kind,'independent-inert-census-actual-proof-observations');assert.equal(numerical.acceptedFiniteNumericalProof,true);assert.equal(numerical.canonical,146);assert.equal(numerical.targeted,31);assert.equal(numerical.keyWitnesses,11);assert.equal(numerical.allFourActualIdentitiesAbsent,true);
 const reporting=pin(m.reportingControls);assert.equal(reporting.kind,'query-census-inert-reporting-controls-v1');assert.equal(reporting.names.length,12);assert.equal(new Set(reporting.names).size,12);assert.equal(reporting.physicalImports,false);assert.equal(reporting.publicVitestLifecycleExercised,false);assert.equal(reporting.sourceCoupledInertCallbacks,true);assert.equal(reporting.acceptance,false);
 const reportingOwner=pin(m.reportingOwner);assert.equal(reportingOwner.kind,'query-census-reporting-result-v1');assert.equal(reportingOwner.completed,true);assert.equal(reportingOwner.status,0);assert.equal(reportingOwner.primary,null);assert.deepEqual(reportingOwner.postErrors,[]);assert.equal(reportingOwner.processes.length,1);assert.equal(reportingOwner.processes[0].code,0);assert.equal(reportingOwner.processes[0].receipt.exit.code,0);assert.equal(reportingOwner.processes[0].receipt.exit.signal,null);assert.equal(reportingOwner.processes[0].receipt.failure,null);assert.deepEqual(reportingOwner.processes[0].receipt.cleanup.executing,[]);
 assert.equal(approval.reportingOwnerSha256,m.reportingOwner.sha256);pin(m.authority);
 assert.equal(m.entry.path,resolve(m.setup,'entry.ts'));assert.deepEqual(Object.keys(m.virtualSources).sort(),['src/core/physics/damage-controls.ts','src/core/physics/damage-thermal.ts']);for(const [file,row]of Object.entries(m.virtualSources)){assert.equal(row.path,resolve(m.setup,file.endsWith('damage-controls.ts')?'damage-controls.census.ts.txt':'damage-thermal.census.ts.txt'));assert.equal(sha(row.path),row.sha256);}
 assert.deepEqual(approval.virtualSourceHashes,Object.fromEntries(Object.entries(m.virtualSources).map(([file,row])=>[file,row.sha256])));assert.equal(approval.entrySha256,m.entry.sha256);
 assert.deepEqual(approval.graph,m.graph);assert.equal(Object.keys(m.graph).length,55);assert.deepEqual(approval.originalSources,m.originalSources);
 assert.deepEqual(approval.workerConditions,m.workerConditions);assert.equal(new Set(m.workerConditions).size,m.workerConditions.length);for(const condition of m.workerConditions)assert.ok(['module','node','development','production'].includes(condition));assert.ok(!(m.workerConditions.includes('development')&&m.workerConditions.includes('production')));assert.deepEqual(approval.prototypeAndIntrinsicDomain,['closed-source-owned-data','no-foreign-callbacks','no-spies-mocks-preload-or-intrinsic-mutation']);
 assert.equal(sha(m.utility.path),m.utility.sha256);const loader=await import(pathToFileURL(m.utility.path).href);
 process.env.npm_execpath=m.npmCli;const authority=parse(m.authority.path);
 assert.deepEqual(loader.sourceSnapshot(m.root),authority.source);assert.deepEqual(loader.installedToolSnapshot(m.root),authority.tools);
 assert.equal(m.testTimeout,30000);assert.equal(m.originalCaseNames.length,3);assert.equal(m.originalSources['tests/golden/booster-replay.test.ts'],sha(resolve(m.root,'tests/golden/booster-replay.test.ts')));
 return {m,loader};
}
// Generated compiler class is separate from the exact mapped/core source closure.
export function certifyGeneratedRuntime(m,edges,directory){
 const spec=pin(m.generatedRuntime);assert.equal(spec.kind,'closed-three-exact-generated-runtime-class-v1');assert.equal(spec.compilerModuleId,'\0rolldown/runtime.js');assert.equal(spec.exactlyOneGeneratedModule,true);assert.equal(spec.mappedSourceCredit,false);assert.equal(spec.helperStart,0);assert.equal(spec.helperEndExclusive,233);assert.equal(Buffer.byteLength(spec.helperText),233);assert.equal(createHash('sha256').update(spec.helperText).digest('hex'),spec.helperSha256);
 const carriers=edges.filter(edge=>edge.moduleIds.includes(spec.compilerModuleId));assert.equal(carriers.length,1);const carrier=carriers[0];assert.equal(carrier.isEntry,false);assert.equal(carrier.moduleIds.filter(id=>id===spec.compilerModuleId).length,1);
 const markers=[];for(const edge of edges){const code=readFileSync(resolve(directory,edge.fileName));const text=code.toString('utf8'),marker='//#region '+String.fromCharCode(92)+'0rolldown/runtime.js';const count=text.split(marker).length-1;if(count){assert.equal(count,1);assert.equal(edge.fileName,carrier.fileName);assert.equal(text.indexOf(marker),0);assert.equal(code.subarray(0,233).toString('utf8'),spec.helperText);assert.equal(createHash('sha256').update(code.subarray(0,233)).digest('hex'),spec.helperSha256);markers.push(edge.fileName);}}
 assert.deepEqual(markers,[carrier.fileName]);for(const[file,h]of Object.entries(spec.producerFiles)){assert.equal(m.selected[file],h);assert.equal(sha(file),h);}
 return{kind:'closed-three-generated-runtime-observation-v1',compilerModuleId:spec.compilerModuleId,fileName:carrier.fileName,certificateSha256:m.generatedRuntime.sha256,helperStart:0,helperEndExclusive:233,helperSha256:spec.helperSha256,class:spec.class,producerFiles:spec.producerFiles,mappedSourceCredit:false};
}
export async function admitArtifacts(path,expected){
 assert.equal(sha(path),expected);const artifact=parse(path);assert.equal(artifact.kind,'closed-three-original-query-census-artifact-v1');assert.equal(artifact.freshBuildCompleted,true);
 const {m,loader}=await admitSource(artifact.sourceInput.path,artifact.sourceInput.sha256);
 assert.equal(artifact.directory,resolve(m.setup,'native'));assert.equal(artifact.receipt,m.setup);
 const actual=loader.filesUnder(artifact.directory).map(file=>relative(artifact.directory,file)).sort();assert.deepEqual(actual,Object.keys(artifact.files).sort());
 for(const [file,digest]of Object.entries(artifact.files))assert.equal(sha(resolve(artifact.directory,file)),digest);
 assert.deepEqual(artifact.graph,m.graph);assert.deepEqual(artifact.virtualSources,m.virtualSources);assert.equal(artifact.entrySha256,m.entry.sha256);
 assert.equal(artifact.sourceInputsSha256,m.sourceInputs.sha256);assert.equal(artifact.bodyApprovalSha256,m.bodyApproval.sha256);
 assert.ok(artifact.edges.length>0&&artifact.edges.length<=16);const owned=new Set(artifact.edges.map(row=>row.fileName));assert.equal(owned.size,artifact.edges.length);
 const ids=new Set();for(const edge of artifact.edges){assert.equal(artifact.files[edge.fileName],edge.sha256);assert.deepEqual(edge.dynamicImports,[]);for(const imported of edge.imports)assert.ok(owned.has(imported));for(const id of edge.moduleIds)ids.add(id);}
 const generatedRuntime=certifyGeneratedRuntime(m,artifact.edges,artifact.directory);assert.deepEqual(generatedRuntime,artifact.generatedRuntime);assert.deepEqual([...ids].sort(),[...Object.keys(m.graph).map(file=>resolve(m.root,file)),m.entry.path,'\0rolldown/runtime.js'].sort());
 const entry=artifact.edges.find(row=>row.fileName==='entry.mjs');assert.equal(entry.isEntry,true);assert.equal(entry.facadeModuleId,m.entry.path);
 const mapped={};for(const file of actual.filter(file=>file.endsWith('.map'))){const map=parse(resolve(artifact.directory,file));assert.equal(map.sources.length,map.sourcesContent.length);
  for(let i=0;i<map.sources.length;i++){const source=resolve(dirname(resolve(artifact.directory,file)),map.sourceRoot??'',map.sources[i]);const relativeName=relative(m.root,source);const virtual=m.virtualSources[relativeName];const expectedText=virtual?readFileSync(virtual.path,'utf8'):readFileSync(source,'utf8');assert.equal(map.sourcesContent[i],expectedText);mapped[source]=sha(virtual?.path??source);}}
 const expectedMapped={...Object.fromEntries(Object.entries(m.graph).map(([file,h])=>[resolve(m.root,file),m.virtualSources[file]?.sha256??h])),[m.entry.path]:m.entry.sha256};
 assert.deepEqual(mapped,expectedMapped);return{m,artifact};
}

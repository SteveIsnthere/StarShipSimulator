// BEGIN exact original-parent validator
function identifyOriginalParent(url, originals, tokenDigits) {
  if (typeof url !== 'string') return null;
  for (const canonicalParentURL of originals.keys()) {
    if (url === canonicalParentURL) return { canonicalParentURL, cacheToken: null };
    const prefix = canonicalParentURL + '?vitest=';
    if (!url.startsWith(prefix)) continue;
    const token = url.slice(prefix.length);
    if (new RegExp('^[0-9]{' + tokenDigits + '}$').test(token)
      && Number.isSafeInteger(Number(token)) && String(Number(token)) === token) {
      return { canonicalParentURL, cacheToken: token };
    }
  }
  return null;
}
// END exact original-parent validator
function sourceBytes(result) {
  assert(typeof result.source === 'string' || ArrayBuffer.isView(result.source) || result.source instanceof ArrayBuffer);
  const source = typeof result.source === 'string' ? Buffer.from(result.source, 'utf8')
    : result.source instanceof ArrayBuffer ? Buffer.from(result.source)
    : Buffer.from(result.source.buffer, result.source.byteOffset, result.source.byteLength);
  assert(source.length <= 16 * 1024 * 1024);
  return source;
}

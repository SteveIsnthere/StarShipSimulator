import type { SimState } from '$core/state';
let __originalStepCalls = 0;
export function step(previous: SimState, dt: number, input: StepInput = NO_INPUT, model: VehicleDefinition = defaultShip): SimState {
  __originalStepCalls++;
  return originalStep(previous, dt, input, model);
}
export function resetOriginalStepCensus(): void { __originalStepCalls = 0; }
export function readOriginalStepCensus(): number { return __originalStepCalls; }

const digest=file=>createHash('sha256').update(readFileSync(file)).digest('hex');
const m=JSON.parse(readFileSync(input,'utf8'));

assert.match(m.setup,/^\/tmp\/starship-native-golden-policy-pure\.[a-f0-9]{32}$/);
assert.equal(fileURLToPath(import.meta.url),resolve(m.setup,'owner.mjs'));
const save=(name,value)=>writeFileSync(resolve(m.setup,name),JSON.stringify(value,null,2)+'\n',{flag:'wx'});
function snapshot(){
 const rows={};for(const [file,sha]of Object.entries(m.inputs)){rows[file]=digest(file);assert.equal(rows[file],sha,file);}
 for(const [file,row]of Object.entries(m.executed)){rows[file]=digest(file);assert.equal(rows[file],row.sha256,file);}
 rows[process.execPath]=digest(process.execPath);assert.equal(rows[process.execPath],m.node.sha256);return rows;
}

 {name:'native-golden-helper-controls',file:'helper-controls.mjs',result:'helper-result.json',kind:'native-golden-support-policy-controls-v1',count:38},
 {name:'native-golden-hook-controls',file:'hook-controls.mjs',result:'hook-result.json',kind:'native-golden-exact-inert-hook-controls-v1',count:28},
];
let before,code=null,primary=null,absence=null;let verifyNoOwnedProcesses;
const processReceipts={},outputs={};

 const owned=await import(pathToFileURL(m.helper.path).href);verifyNoOwnedProcesses=owned.verifyNoOwnedProcesses;

  if(program.count===28)args.push(resolve(m.setup,'loader.mjs'),m.executed[resolve(m.setup,'loader.mjs')].sha256);
  args.push(resolve(m.setup,program.result));

   deadlineMs:Math.min(5000,22000-(Date.now()-started)-8000),env:{...process.env,NODE_OPTIONS:''}});
  // Helper raw exit is durable before any result interpretation or next child.
  const raw=JSON.parse(readFileSync(resolve(m.setup,program.name+'.process.json'),'utf8'));
  processReceipts[program.name]=raw;save(program.name+'.actual-status.json',{code,exit:raw.exit,failure:raw.failure});
  assert.equal(code,0);assert.equal(raw.exit.code,0);assert.equal(raw.exit.signal,null);assert.equal(raw.failure,null);
  assert.equal(raw.closed,true);assert.equal(raw.finished,true);assert.equal(raw.cleanup.executing.length,0);assert(raw.cleanup.tracked.length>0);
  assert.equal(readFileSync(resolve(m.setup,program.name+'.stderr.txt')).length,0);
  const stdout=readFileSync(resolve(m.setup,program.name+'.stdout.txt'),'utf8');assert.equal(stdout.trim().split('\n').length,1);
  const summary=JSON.parse(stdout);assert.equal(summary.completed,true);assert.equal(summary.cases,program.count);
  const result=JSON.parse(readFileSync(resolve(m.setup,program.result),'utf8'));outputs[program.name]=result;
  assert.equal(result.kind,program.kind);assert.equal(result.completed,true);assert.equal(result.code,0);assert.equal(result.failure,null);assert.deepEqual(result.afterErrors,[]);
  assert.equal(result.cases.length,program.count);assert.equal(new Set(result.cases).size,program.count);

 }
}catch(error){primary=String(error.stack??error);}

}

save('owner-result.json',{kind:'native-golden-policy-owner-result-v1',completed,rawCode:code,failure:primary,processReceipts,outputs,absence,elapsedMs,
 pureAndInertCallbacksOnly:true,sourceOrInstalledQualification:false,parentMustIndependentlyVerifyALLTrackedAndObservedPidStarts:true});
process.stdout.write(resolve(m.setup,'owner-result.json')+'\n');process.exitCode=completed?0:1;

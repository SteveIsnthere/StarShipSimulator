import {registerHooks, stripTypeScriptTypes} from 'node:module';
import {readFileSync,existsSync} from 'node:fs';
import {fileURLToPath,pathToFileURL} from 'node:url';
const root='/Users/stevewang/dev/StarShipSimulator-realism';
registerHooks({resolve(specifier,context,next){
 let s=specifier;
 const alias=s.match(/^\$(core|app|ui|view|hud|audio)\/(.*)/);
 if(alias)s=pathToFileURL(`${root}/src/${alias[1]}/${alias[2]}`).href;
 if(s.startsWith('file:') || s.startsWith('.')){
 const u=new URL(s,context.parentURL);
 const p=fileURLToPath(u);
 if(!existsSync(p) && existsSync(p+'.ts'))s=u.href+'.ts';
 if(existsSync(p+'/index.ts'))s=u.href+'/index.ts';
 }
 return next(s,context);
},load(url,context,next){
 if(url.startsWith('file:') && url.endsWith('.ts'))return {format:'module',source:stripTypeScriptTypes(readFileSync(fileURLToPath(url),'utf8').replaceAll('import.meta.env.DEV','false')),shortCircuit:true};
 return next(url,context);
}});
globalThis.window={matchMedia:()=>({matches:true}),localStorage:{getItem:()=>null}};
const {createSession}=await import(`${root}/src/ui/session/session.ts`);
const {getScenario}=await import(`${root}/src/core/scenarios.ts`);
const {execFileSync}=await import('node:child_process');
const {default:assert}=await import('node:assert/strict');
const target='85a302db1237c9e71e3294fa3ba969c98edfc589';
assert.equal(execFileSync('git',['rev-parse','HEAD'],{cwd:root,encoding:'utf8'}).trim(),target);
console.log('source',target);
const DT=1/120;
function started(){const s=createSession();s.startFlight(getScenario('rtls'));s.emit({type:'autoLand'});s.advance(DT);return s;}
const session=started(),job=session.loop.state.autopilot.boosterPrediction;
assert.equal(job.origin.failures.randomFailure,false);
session.toggleRandomFailure();
assert.equal(session.loop.state.failures.randomFailure,true);
assert.equal(session.loop.state.autopilot.boosterPrediction,job);
session.advance(DT);
assert.equal(session.loop.state.autopilot.boosterPrediction.origin.failures.randomFailure,false);
console.log('pending preference change retains old source', {live:true,origin:false});
for(let i=0;i<3000 && !session.loop.state.autopilot.boosterReturnPlan;i++)session.advance(DT);
assert.ok(session.loop.state.autopilot.boosterReturnPlan);
const a=session.loop.state.autopilot;
console.log('actual stale-source publication',JSON.stringify({time:session.loop.state.world.environmentTime,live:session.loop.state.failures.randomFailure,origin:a.boosterPrediction.origin.failures.randomFailure,plan:a.boosterReturnPlan,failedEngineIndices:session.loop.state.engines.failed.flatMap((f,i)=>f?[i]:[])}));
const oldPlan=a.boosterReturnPlan;
session.toggleRandomFailure();
assert.equal(session.loop.state.failures.randomFailure,false);
assert.equal(session.loop.state.autopilot.boosterReturnPlan,oldPlan);
console.log('accepted plan survives subsequent preference change',true);
session.emit({type:'pitch',percent:1});
assert.equal(session.loop.state.autopilot.boosterPrediction,undefined);
assert.equal(session.loop.state.autopilot.boosterReturnPlan,undefined);
session.advance(DT);
assert.equal(session.loop.state.autopilot.boosterPrediction.origin.failures.randomFailure,false);
console.log('positive control: manual event clears authority and next origin matches live',true);
const fixed=started();fixed.toggleRandomFailure();fixed.emit({type:'throttle',percent:fixed.readThrottle()});fixed.advance(DT);
assert.equal(fixed.loop.state.autopilot.boosterPrediction.origin.failures.randomFailure,true);
console.log('positive control: explicit invalidation after toggle takes new policy source',true);
const unchanged=started(),oldOrigin=unchanged.loop.state.autopilot.boosterPrediction.origin;
unchanged.advance(DT);
assert.equal(unchanged.loop.state.autopilot.boosterPrediction.origin,oldOrigin);
assert.equal(oldOrigin.failures.randomFailure,unchanged.loop.state.failures.randomFailure);
console.log('negative control: unchanged policy safely retains origin',true);
const mission=createSession();mission.startHotStage(123);mission.stage();
for(let i=0;i<240 && mission.mission.phase==='attached';i++)mission.advance(DT);
assert.equal(mission.mission.phase,'separated');
mission.selectVehicle('super-heavy');mission.emit({type:'autoLand'});mission.advance(DT);
const boosterJob=mission.mission.booster.autopilot.boosterPrediction;
mission.selectVehicle('ship');mission.toggleRandomFailure();
assert.equal(mission.mission.booster.failures.randomFailure,true);
assert.equal(mission.mission.booster.autopilot.boosterPrediction,boosterJob);
assert.equal(boosterJob.origin.failures.randomFailure,false);
console.log('unselected mission booster also retains old policy source',true);
